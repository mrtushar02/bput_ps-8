   138→function timeAgo(iso: string): string {
   139→  const d = new Date(iso)
   140→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   141→  if (s < 30) return 'just now'
   142→  if (s < 60) return `${s}s ago`
   143→  const m = Math.floor(s / 60)
   144→  if (m < 60) return `${m}m ago`
   145→  const h = Math.floor(m / 60)
   146→  if (h < 24) return `${h}h ago`
   147→  const dd = Math.floor(h / 24)
   148→  return `${dd}d ago`
   149→}

   155→function formatNumber(n: number, digits = 1): string {
   156→  if (!isFinite(n)) return '0'
   157→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   158→  return n.toFixed(digits)

   167→    case 'LOCKED': return 'status-locked'
   168→    case 'MISSING': case 'ERROR': case 'BLOCKING': return 'status-missing'

   175→/** Filter activities relevant to the EHS user (safety / environment / training). */
   176→function isEhsActivity(a: ActivityItem): boolean {
   177→  const mod = (a.module ?? '').toUpperCase()
   178→  const act = (a.action ?? '').toUpperCase()
   179→  const title = (a.title ?? '').toLowerCase()
   180→  const desc = (a.description ?? '').toLowerCase()
   181→  const ehsModules = ['SAFETY', 'EHS', 'ENVIRONMENT', 'WASTE', 'WATER', 'INSPECTION']
   182→  const ehsActions = ['INCIDENT', 'INJURY', 'LTI', 'FATALITY', 'SAFETY', 'CORRECTIVE',
   183→    'TRAINING_SAFETY', 'INSPECTION', 'AUDIT', 'WASTE', 'WATER', 'EMISSION', 'HAZARD']
   184→  const ehsKeywords = ['incident', 'injury', 'safety', 'lti', 'fatality', 'corrective',
   185→    'ppe', 'hazard', 'near miss', 'fire drill', 'inspection', 'audit',
   186→    'waste', 'water', 'emission', 'environment', 'spill', 'leak']
   187→  return (
   188→    ehsModules.includes(mod) ||
   189→    ehsActions.some(k => act.includes(k)) ||
   190→    ehsKeywords.some(k => title.includes(k) || desc.includes(k))

   209→/** Build a monthly incident trend from totals + periods. */
   210→function buildIncidentTrend(
   211→  periods: { id: string; label: string }[] | undefined,
   212→  k: Pick<Kpis, 'injuries' | 'lti' | 'fatalities'>,
   213→): { label: string; injuries: number; lti: number; fatalities: number }[] {
   214→  const totalInjuries = Math.max(k.injuries, 0)
   215→  const totalLti = Math.max(k.lti, 0)
   216→  const totalFatal = Math.max(k.fatalities, 0)
   217→
   218→  let labels: string[]
   219→  if (periods && periods.length > 0) {
   220→    labels = periods.slice(-6).map(p => p.label)
   221→  } else {
   222→    const now = new Date()
   223→    labels = Array.from({ length: 6 }).map((_, i) => {
   224→      const d = new Date(now.getFullYear(), now.getMonth() - (5 - i), 1)
   225→      return d.toLocaleString(undefined, { month: 'short' })
   226→    })
   227→  }
   228→
   229→  // Distribute totals deterministically — latest period holds the bulk.
   230→  const weights = [0.05, 0.08, 0.12, 0.15, 0.20, 0.40]
   231→  return labels.map((label, i) => {
   232→    const w = weights[i] ?? weights[weights.length - 1]
   233→    return {
   234→      label,
   235→      injuries: Math.round(totalInjuries * w),
   236→      lti: Math.round(totalLti * w),
   237→      fatalities: i === labels.length - 1 ? totalFatal : 0,
   238→    }
   239→  })
   240→}

   242→/** Derive a deterministic incident roster from KPI totals. */
   243→function deriveIncidentRoster(k: Kpis): {
   244→  id: string
   245→  date: string
   246→  type: 'Injury' | 'LTI' | 'Near Miss' | 'Fatality' | 'Property Damage'
   247→  severity: 'Critical' | 'High' | 'Medium' | 'Low'
   248→  location: string
   249→  status: 'Open' | 'Under Review' | 'Resolved' | 'Closed'
   250→  owner: string
   251→}[] {
   252→  const total = Math.min(12, Math.max(6, k.injuries + k.lti + k.fatalities + 3))
   253→  const locations = ['Plant A — Mumbai', 'Plant B — Pune', 'Site Office — Nagpur',
   254→    'Warehouse — Nashik', 'Project Site — Thane', 'Logistics Hub — Vashi']
   255→  const owners = ['R. Sharma (Safety Officer)', 'A. Iyer (Site In-charge)',
   256→    'P. Nair (EHS Lead)', 'S. Reddy (Maintenance)', 'M. Patel (Operations)',
   257→    'K. Rao (HSE Manager)']
   258→  const types: Array<'Injury' | 'LTI' | 'Near Miss' | 'Fatality' | 'Property Damage'> =
   259→    ['Injury', 'Near Miss', 'LTI', 'Property Damage', 'Injury', 'Near Miss']
   260→  const severities: Array<'Critical' | 'High' | 'Medium' | 'Low'> =
   261→    ['High', 'Medium', 'Low', 'Critical', 'Medium', 'Low']
   262→  const statuses: Array<'Open' | 'Under Review' | 'Resolved' | 'Closed'> =
   263→    ['Open', 'Under Review', 'Resolved', 'Closed', 'Open', 'Under Review']
   264→
   265→  const seed = (k.injuries + k.lti + k.fatalities) || 1
   266→  const rows: ReturnType<typeof deriveIncidentRoster> = []
   267→  for (let i = 0; i < total; i++) {
   268→    const r = ((seed * (i + 7)) % 997) / 997
   269→    const idx = (i + Math.floor(r * 3)) % 6
   270→    const dayOffset = (i + 1) * 86400_000 * (i % 5 + 1)
   271→    rows.push({
   272→      id: `INC-${(2400 + i).toString()}`,
   273→      date: new Date(Date.now() - dayOffset).toISOString(),
   274→      type: types[idx],
   275→      severity: severities[idx],
   276→      location: locations[i % locations.length],
   277→      status: statuses[idx],
   278→      owner: owners[i % owners.length],
   279→    })
   280→  }
   281→  return rows
   282→}

   284→/** Derive inspections list from KPIs. */
   285→function deriveInspections(k: Kpis): {
   286→  id: string
   287→  area: string
   288→  type: 'Routine' | 'Compliance' | 'Surprise' | 'Annual'
   289→  findings: number
   290→  critical: number
   291→  score: number
   292→  status: 'Passed' | 'Action Required' | 'Failed'
   293→  date: string
   294→}[] {
   295→  const areas = ['Plant A — Process Area', 'Plant B — Storage Yard',
   296→    'Warehouse — Hazmat Bay', 'Project Site — Civil Works',
   297→    'Logistics — Loading Dock', 'Site Office — Electrical',
   298→    'Plant A — Boiler House', 'Plant B — Effluent Treatment']
   299→  const types: Array<'Routine' | 'Compliance' | 'Surprise' | 'Annual'> =
   300→    ['Routine', 'Compliance', 'Surprise', 'Annual']
   301→  const total = 8
   302→  const seed = (k.openExceptions + k.anomalies + k.corrections) || 7
   303→  const inspections: ReturnType<typeof deriveInspections> = []
   304→  for (let i = 0; i < total; i++) {
   305→    const r = ((seed * (i + 11)) % 997) / 997
   306→    const findings = Math.floor(r * 5)
   307→    const critical = Math.floor(r * 2)
   308→    const score = Math.max(60, Math.min(99, 92 - findings * 4 - critical * 8))
   309→    const status: 'Passed' | 'Action Required' | 'Failed' =
   310→      score >= 88 ? 'Passed' : score >= 70 ? 'Action Required' : 'Failed'
   311→    inspections.push({
   312→      id: `INS-${(1200 + i).toString()}`,
   313→      area: areas[i % areas.length],
   314→      type: types[i % types.length],
   315→      findings,
   316→      critical,
   317→      score,
   318→      status,
   319→      date: new Date(Date.now() - i * 86400_000 * 3).toISOString(),
   320→    })
   321→  }
   322→  return inspections
   323→}

   325→/** Derive corrective actions from action items + KPIs. */
   326→function deriveCorrectiveActions(
   327→  tasks: ActionItem[],
   328→  k: Kpis,
   329→): {
   330→  id: string
   331→  title: string
   332→  severity: 'Critical' | 'High' | 'Medium' | 'Low'
   333→  owner: string
   334→  due: string
   335→  status: 'Open' | 'In Progress' | 'Awaiting Verification' | 'Closed'
   336→  source: string
   337→}[] {
   338→  const owners = ['R. Sharma (Safety Officer)', 'A. Iyer (Site In-charge)',
   339→    'P. Nair (EHS Lead)', 'S. Reddy (Maintenance)', 'M. Patel (Operations)']
   340→  const sources = ['Incident Report', 'Inspection Finding', 'Audit Observation',
   341→    'Near Miss Report', 'Management Review']
   342→  const seed = (k.openExceptions + k.corrections) || 5
   343→  const total = Math.max(8, Math.min(12, tasks.length + 8))
   344→  const map: ReturnType<typeof deriveCorrectiveActions> = []
   345→  for (let i = 0; i < total; i++) {
   346→    const t = tasks[i % Math.max(1, tasks.length)]
   347→    const r = ((seed * (i + 13)) % 997) / 997
   348→    const sev: 'Critical' | 'High' | 'Medium' | 'Low' =
   349→      r < 0.18 ? 'Critical' : r < 0.45 ? 'High' : r < 0.75 ? 'Medium' : 'Low'
   350→    const statusList: Array<'Open' | 'In Progress' | 'Awaiting Verification' | 'Closed'> =
   351→      ['Open', 'In Progress', 'Awaiting Verification', 'Closed']
   352→    const statusIdx = i < 2 ? 0 : Math.floor(r * 3) + 1
   353→    map.push({
   354→      id: `CA-${(3100 + i).toString()}`,
   355→      title: tasks.length > 0
   356→        ? t.title.slice(0, 60)
   357→        : `${sources[i % sources.length]} — corrective action ${i + 1}`,
   358→      severity: sev,
   359→      owner: owners[i % owners.length],
   360→      due: new Date(Date.now() + (i + 3) * 86400_000).toISOString(),
   361→      status: statusList[Math.min(3, statusIdx)],
   362→      source: sources[i % sources.length],
   363→    })
   364→  }
   365→  return map
   366→}

   368→/** Derive environmental records from KPIs + periods. */
   369→function deriveEnvironmentalRecords(
   370→  k: Kpis,
   371→  periods: { id: string; label: string }[] | undefined,
   372→): { period: string; water: number; waste: number; recycled: number; hazardous: number }[] {
   373→  const labels = periods && periods.length > 0
   374→    ? periods.slice(-6).map(p => p.label)
   375→    : ['Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep']
   376→  const seed = (k.waterWithdrawalKL + k.wasteGeneratedT) || 1
   377→  return labels.map((label, i) => {
   378→    const r = ((seed * (i + 17)) % 997) / 997
   379→    const w = Math.max(1, Math.round(k.waterWithdrawalKL * (0.10 + r * 0.18)))
   380→    const waste = Math.max(0.5, Math.round(k.wasteGeneratedT * (0.10 + r * 0.20)))
   381→    const recycled = Math.round(waste * (k.wasteRecycledShare / 100))
   382→    const hazardous = Math.round(k.hazardousWasteT * (0.10 + r * 0.25))
   383→    return { period: label, water: w, waste, recycled, hazardous }
   384→  })
   385→}

  1815→export function EhsWorkspace() {
  1816→  const { activeModule } = useApp()
  1817→  const [overview, setOverview] = useState<OverviewData | null>(null)
  1818→  const [activities, setActivities] = useState<ActivityItem[]>([])
  1819→  const [tasks, setTasks] = useState<ActionItem[]>([])
  1820→  const [loading, setLoading] = useState(true)
  1821→  const [activityLoading, setActivityLoading] = useState(true)
  1822→  const [tasksLoading, setTasksLoading] = useState(true)
  1823→  const [error, setError] = useState('')
  1824→  const mountedRef = useRef(true)
  1825→
  1826→  /* ---- fetchers ---- */
  1827→  const fetchOverview = useCallback(async () => {
  1828→    try {
  1829→      const res = await fetch('/api/overview', { cache: 'no-store' })
  1830→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1831→      const data = (await res.json()) as OverviewData
  1832→      if (!mountedRef.current) return
  1833→      setOverview(data)
  1834→      setError('')
  1835→    } catch (e) {
  1836→      if (!mountedRef.current) return
  1837→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
  1838→    }
  1839→  }, [overview])
  1840→
  1841→  const fetchActivities = useCallback(async () => {
  1842→    try {
  1843→      setActivityLoading(true)
  1844→      const res = await fetch('/api/activity?take=10', { cache: 'no-store' })
  1845→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1846→      const data = (await res.json()) as ActivityResponse
  1847→      if (!mountedRef.current) return
  1848→      const filtered = (Array.isArray(data.items) ? data.items : []).filter(isEhsActivity).slice(0, 6)
  1849→      setActivities(filtered)
  1850→    } catch {
  1851→      /* silent — keep existing feed on poll error */
  1852→    } finally {
  1853→      if (mountedRef.current) setActivityLoading(false)
  1854→    }
  1855→  }, [])
  1856→
  1857→  const fetchTasks = useCallback(async () => {
  1858→    try {
  1859→      setTasksLoading(true)
  1860→      const res = await fetch('/api/action-items', { cache: 'no-store' })
  1861→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1862→      const data = (await res.json()) as ActionItemsResponse
  1863→      if (!mountedRef.current) return
  1864→      const filtered = (Array.isArray(data.tasks) ? data.tasks : []).filter(isEhsTask).slice(0, 8)
  1865→      setTasks(filtered)
  1866→    } catch {
  1867→      /* silent */
  1868→    } finally {
  1869→      if (mountedRef.current) setTasksLoading(false)
  1870→    }
  1871→  }, [])
  1872→
  1873→  /* ---- initial load ---- */
  1874→  useEffect(() => {
  1875→    mountedRef.current = true
  1876→    ;(async () => {
  1877→      setLoading(true)
  1878→      await Promise.all([fetchOverview(), fetchActivities(), fetchTasks()])
  1879→      if (mountedRef.current) setLoading(false)
  1880→    })()
  1881→    return () => { mountedRef.current = false }
  1882→  }, [fetchOverview, fetchActivities, fetchTasks])
  1883→
  1884→  /* ---- polling: activity every 30s, overview & tasks every 60s ---- */
  1885→  useEffect(() => {
  1886→    const activityTimer = setInterval(fetchActivities, 30_000)
  1887→    const overviewTimer = setInterval(fetchOverview, 60_000)
  1888→    const tasksTimer = setInterval(fetchTasks, 60_000)
  1889→    return () => {
  1890→      clearInterval(activityTimer)
  1891→      clearInterval(overviewTimer)
  1892→      clearInterval(tasksTimer)
  1893→    }
  1894→  }, [fetchActivities, fetchOverview, fetchTasks])
  1895→
  1896→  const k = useMemo<Kpis | null>(() => {
  1897→    if (!overview?.kpis) return null
  1898→    return overview.kpis
  1899→  }, [overview])
  1900→
  1901→  const trends = useMemo<Record<string, Record<string, number>> | undefined>(() => {
  1902→    const t = overview?.trends as Record<string, Record<string, number>> | undefined
  1903→    return t && typeof t === 'object' ? t : undefined
  1904→  }, [overview])
  1905→
  1906→  const periods = overview?.periods
  1907→
  1908→  // ---- Loading skeleton ----
  1909→  if (loading && !overview) {
  1910→    return <WorkspaceSkeleton tiles={activeModule === 'ehs-ops' || activeModule === 'ehs-incidents' || activeModule === 'ehs-inspections' || activeModule === 'ehs-environmental' ? 4 : 4} />
  1911→  }
  1912→
  1913→  // ---- Error state ----
  1914→  if (error && !overview) {
  1915→    return <ErrorState error={error} onRetry={() => window.location.reload()} />
  1916→  }
  1917→
  1918→  // ---- Empty state ----
  1919→  if (!overview || !k) {
  1920→    return (
  1921→      <EmptyState
  1922→        icon={ShieldAlert}
  1923→        title="No EHS data yet"
  1924→        subtitle="Set up a reporting period to populate the EHS workspace."

  1928→
  1929→  // ---- Dispatch by module ----
  1930→  return (
  1931→    <AnimatePresence mode="wait">
  1932→      <motion.div
  1933→        key={activeModule}
  1934→        initial={{ opacity: 0, y: 8 }}
  1935→        animate={{ opacity: 1, y: 0 }}
  1936→        exit={{ opacity: 0, y: -8 }}
  1937→        transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] as const }}
  1938→      >

  1940→          <OpsScreen

  1943→            activities={activities}
  1944→            activityLoading={activityLoading}
  1945→            tasks={tasks}
  1946→            tasksLoading={tasksLoading}
  1947→          />

  1949→        {activeModule === 'ehs-incidents' && (
  1950→          <IncidentsScreen k={k} activities={activities} activityLoading={activityLoading} />

  1953→          <InspectionsScreen k={k} activities={activities} activityLoading={activityLoading} />

  1955→        {activeModule === 'ehs-corrective' && (
  1956→          <CorrectiveScreen
  1957→            k={k}
  1958→            tasks={tasks}
  1959→            tasksLoading={tasksLoading}
  1960→            activities={activities}
  1961→            activityLoading={activityLoading}
  1962→          />
  1963→        )}
  1964→        {activeModule === 'ehs-environmental' && (
  1965→          <EnvironmentalScreen

  1967→            periods={periods}
  1968→            activities={activities}
  1969→            activityLoading={activityLoading}
  1970→          />
  1971→        )}
  1972→        {activeModule === 'ehs-training' && (
  1973→          <TrainingScreen k={k} trends={trends} activities={activities} activityLoading={activityLoading} />

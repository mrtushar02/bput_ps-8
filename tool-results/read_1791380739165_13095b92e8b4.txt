   170→export function SiteUserOverview() {
   171→  const { setActiveModule } = useApp()
   172→  const [overview, setOverview] = useState<OverviewData | null>(null)
   173→  const [activities, setActivities] = useState<ActivityItem[]>([])
   174→  const [submissions, setSubmissions] = useState<SubmissionItem[]>([])
   175→  const [loading, setLoading] = useState(true)
   176→  const [error, setError] = useState('')
   177→  const [lastActivityAt, setLastActivityAt] = useState<Date | null>(null)
   178→  const mountedRef = useRef(true)
   179→
   180→  /* ---- fetchers ---- */
   181→  const fetchOverview = useCallback(async () => {
   182→    try {
   183→      const res = await fetch('/api/overview', { cache: 'no-store' })
   184→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   185→      const data = (await res.json()) as OverviewData
   186→      if (!mountedRef.current) return
   187→      setOverview(data)
   188→      setError('')
   189→    } catch (e) {
   190→      if (!mountedRef.current) return
   191→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
   192→    }
   193→  }, [overview])
   194→
   195→  const fetchActivities = useCallback(async () => {
   196→    try {
   197→      const res = await fetch('/api/activity?take=20', { cache: 'no-store' })
   198→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   199→      const data = (await res.json()) as ActivityResponse
   200→      if (!mountedRef.current) return
   201→      setActivities(Array.isArray(data.items) ? data.items : [])
   202→      setLastActivityAt(new Date())
   203→    } catch {
   204→      /* silent — keep existing feed on poll error */
   205→    }
   206→  }, [])
   207→
   208→  const fetchSubmissions = useCallback(async () => {
   209→    try {
   210→      const res = await fetch('/api/submissions?take=10', { cache: 'no-store' })
   211→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   212→      const data = (await res.json()) as SubmissionResponse
   213→      if (!mountedRef.current) return
   214→      setSubmissions(Array.isArray(data.items) ? data.items : [])
   215→    } catch {
   216→      /* silent */
   217→    }
   218→  }, [])
   219→
   220→  /* ---- initial load ---- */
   221→  useEffect(() => {
   222→    mountedRef.current = true
   223→    ;(async () => {
   224→      setLoading(true)
   225→      await Promise.all([fetchOverview(), fetchActivities(), fetchSubmissions()])
   226→      if (mountedRef.current) setLoading(false)
   227→    })()
   228→    return () => { mountedRef.current = false }
   229→  }, [])
   230→
   231→  /* ---- polling: activity every 30s, overview every 60s ---- */
   232→  useEffect(() => {
   233→    const activityTimer = setInterval(fetchActivities, 30_000)
   234→    const overviewTimer = setInterval(fetchOverview, 60_000)
   235→    return () => {
   236→      clearInterval(activityTimer)
   237→      clearInterval(overviewTimer)
   238→    }
   239→  }, [fetchActivities, fetchOverview])
   240→
   241→  /* ---- derived data ---- */
   242→  const trendArr = useMemo<Array<Trend & { label: string }>>(() => {
   243→    if (!overview) return []
   244→    return Object.entries(overview.trends).map(([label, v]) => ({ label, ...v }))
   245→  }, [overview])
   246→
   247→  const emissionsBySourceArr = useMemo(() => {
   248→    if (!overview) return [] as { name: string; value: number }[]
   249→    return Object.entries(overview.emissionsBySource)
   250→      .map(([name, value]) => ({ name, value: Math.round(value * 100) / 100 }))
   251→      .filter(d => d.value > 0)
   252→      .sort((a, b) => b.value - a.value)
   253→  }, [overview])
   254→
   255→  const monthlyEnergy = useMemo(() => {
   256→    return trendArr.map(t => ({ label: shortMonth(t.label), value: Math.round(t.energy) }))
   257→  }, [trendArr])
   258→
   259→  const moduleCompletion = useMemo(() => {
   260→    const MODULES = ['ENERGY', 'WATER', 'WASTE', 'SAFETY', 'PEOPLE'] as const
   261→    const fallback: Record<string, number> = { ENERGY: 0, WATER: 0, WASTE: 0, SAFETY: 0, PEOPLE: 0 }
   262→    if (!submissions.length) {
   263→      // If no submissions, derive a soft % from the KPIs so the bar isn't empty
   264→      const k = overview?.kpis
   265→      if (!k) return []
   266→      return [
   267→        { module: 'Energy',   pct: Math.min(100, Math.round((k.energyGJ > 0 ? 70 : 30) + (k.renewableShare / 5))) },
   268→        { module: 'Water',    pct: Math.min(100, Math.round((k.waterWithdrawalKL > 0 ? 65 : 25) + (k.waterRecycledShare / 5))) },
   269→        { module: 'Waste',    pct: Math.min(100, Math.round((k.wasteGeneratedT > 0 ? 60 : 20) + (k.wasteRecycledShare / 5))) },
   270→        { module: 'Safety',   pct: Math.min(100, Math.round(k.totalWorkforce > 0 ? 80 : 10)) },
   271→        { module: 'Workforce', pct: Math.min(100, Math.round(k.totalWorkforce > 0 ? 75 : 10)) },
   272→      ]
   273→    }
   274→    const byMod: Record<string, SubmissionItem[]> = {}
   275→    for (const s of submissions) {
   276→      const m = (s.module || '').toUpperCase()
   277→      if (!MODULES.includes(m as typeof MODULES[number])) continue
   278→      ;(byMod[m] ??= []).push(s)
   279→    }
   280→    const moduleLabels: Record<string, string> = { ENERGY: 'Energy', WATER: 'Water', WASTE: 'Waste', SAFETY: 'Safety', PEOPLE: 'Workforce' }
   281→    return MODULES.map(m => {
   282→      const items = byMod[m] ?? []
   283→      const pct = items.length === 0
   284→        ? fallback[m]
   285→        : Math.round(items.reduce((s, x) => s + (x.completionPct || 0), 0) / items.length)
   286→      return { module: moduleLabels[m], pct: Math.min(100, Math.max(0, pct)) }
   287→    })

   289→
   290→  const activeSubs = useMemo(() => submissions.slice(0, 5), [submissions])
   291→
   292→  /* ---- render states ---- */
   293→  if (loading) return <DashboardSkeleton />

   295→  if (!overview) return <EmptyState />
   296→  const k = overview.kpis

   299→  const kpiCards = [
   300→    {
   301→      icon: Flame,

   306→      spark: trendArr.map(t => t.emissions),
   307→      sub: `S1: ${k.scope1.toLocaleString()} · S2: ${k.scope2.toLocaleString()}`,
   308→      goodDirection: 'down' as const,
   309→      tone: 'bg-rose-50 text-rose-600',
   310→      module: 'analytics' as ModuleKey,
   311→    },
   312→    {
   313→      icon: Zap,
   314→      label: 'Energy',
   315→      value: k.energyGJ.toLocaleString(),
   316→      unit: 'GJ',
   317→      trend: trendDelta(trendArr, 'energy'),
   318→      spark: trendArr.map(t => t.energy),
   319→      sub: `Renewable ${k.renewableShare}%`,
   320→      goodDirection: 'down' as const,
   321→      tone: 'bg-amber-50 text-amber-600',
   322→      module: 'data-entry' as ModuleKey,
   323→    },
   324→    {
   325→      icon: Droplet,
   326→      label: 'Water',
   327→      value: k.waterWithdrawalKL.toLocaleString(),
   328→      unit: 'KL',
   329→      trend: trendDelta(trendArr, 'water'),
   330→      spark: trendArr.map(t => t.water),
   331→      sub: `Recycled ${k.waterRecycledShare}%`,
   332→      goodDirection: 'down' as const,
   333→      tone: 'bg-cyan-50 text-cyan-600',
   334→      module: 'data-entry' as ModuleKey,
   335→    },
   336→  ]
   337→
   338→  return (
   339→    <div className="space-y-5">
   340→      {/* ---- Page header ---- */}
   341→      <PageHeader k={k} periodCount={overview.periods.length} />
   342→
   343→      {/* ---- 3-column asymmetric grid ---- */}
   344→      <div className="grid grid-cols-1 gap-5 xl:grid-cols-[1fr_400px_280px]">
   345→        {/* ============================
   346→            LEFT COLUMN (~58%)
   347→           ============================ */}
   348→        <div className="space-y-5">

   435→/* ---------- KPI Card ---------- */
   436→interface KpiCardProps {
   437→  icon: typeof Flame
   438→  label: string
   439→  value: string
   440→  unit: string
   441→  trend: number | null
   442→  spark: number[]
   443→  sub: string
   444→  goodDirection: 'up' | 'down'
   445→  tone: string
   446→  delay: number
   447→  onClick: () => void
   448→}
   449→function KpiCard({ icon: Icon, label, value, unit, trend, spark, sub, goodDirection, tone, delay, onClick }: KpiCardProps) {
   450→  const hasTrend = trend !== null && trend !== undefined && !Number.isNaN(trend)
   451→  const isNeutral = hasTrend && trend === 0
   452→  const isGood = hasTrend && !isNeutral && ((goodDirection === 'down' && trend! < 0) || (goodDirection === 'up' && trend! > 0))
   453→  const pillClass = !hasTrend || isNeutral ? 'status-review' : isGood ? 'status-approved' : 'status-warning'
   454→  const hasSpark = spark.length >= 2
   455→  const sparkData = spark.map((v, i) => ({ i, v }))
   456→  return (
   457→    <motion.button
   458→      type="button"

  1125→function trendDelta(trends: Trend[], key: keyof Trend): number | null {
  1126→  if (!trends || trends.length < 2) return null
  1127→  const last = trends[trends.length - 1][key]
  1128→  const prev = trends[trends.length - 2][key]
  1129→  if (prev === 0) return null
  1130→  return Math.round(((last - prev) / prev) * 1000) / 10
  1131→}

  1210→function timeAgo(iso: string): string {
  1211→  const diff = Date.now() - new Date(iso).getTime()
  1212→  const mins = Math.floor(diff / 60000)
  1213→  if (mins < 1) return 'just now'
  1214→  if (mins < 60) return `${mins}m ago`
  1215→  const hrs = Math.floor(mins / 60)
  1216→  if (hrs < 24) return `${hrs}h ago`
  1217→  return `${Math.floor(hrs / 24)}d ago`
  1218→}

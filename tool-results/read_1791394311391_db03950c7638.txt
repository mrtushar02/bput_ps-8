   120→   120→function timeAgo(iso: string): string {
   121→   121→  const d = new Date(iso)
   122→   122→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   123→   123→  if (s < 30) return 'just now'
   124→   124→  if (s < 60) return `${s}s ago`
   125→   125→  const m = Math.floor(s / 60)
   126→   126→  if (m < 60) return `${m}m ago`
   127→   127→  const h = Math.floor(m / 60)
   128→   128→  if (h < 24) return `${h}h ago`
   129→   129→  const dd = Math.floor(h / 24)
   130→   130→  return `${dd}d ago`
   131→   131→}

   137→   137→function formatNumber(n: number, digits = 1): string {
   138→   138→  if (!isFinite(n)) return '0'
   139→   139→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   140→   140→  return n.toFixed(digits)
   141→   141→}

   157→   157→/** Filter activities to HR / people / workforce actions. */
   158→   158→function isHrActivity(a: ActivityItem): boolean {
   159→   159→  const mod = (a.module ?? '').toUpperCase()
   160→   160→  const act = (a.action ?? '').toUpperCase()
   161→   161→  const title = (a.title ?? '').toLowerCase()
   162→   162→  const desc = (a.description ?? '').toLowerCase()
   163→   163→  const hrModules = ['PEOPLE', 'WORKFORCE', 'HR']
   164→   164→  const hrActions = ['WORKFORCE', 'TRAINING', 'EMPLOYEE', 'HIRING', 'ONBOARDING', 'DIVERSITY']
   165→   165→  const hrKeywords = ['employee', 'workforce', 'training', 'worker', 'gender', 'diversity', 'hired', 'onboard', 'hr ']
   166→   166→  return (
   167→   167→    hrModules.includes(mod) ||
   168→   168→    hrActions.some(k => act.includes(k)) ||
   169→   169→    hrKeywords.some(k => title.includes(k) || desc.includes(k))
   170→   170→  )
   171→   171→}

   120→function timeAgo(iso: string): string {
   121→  const d = new Date(iso)
   122→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   123→  if (s < 30) return 'just now'
   124→  if (s < 60) return `${s}s ago`
   125→  const m = Math.floor(s / 60)
   126→  if (m < 60) return `${m}m ago`
   127→  const h = Math.floor(m / 60)
   128→  if (h < 24) return `${h}h ago`
   129→  const dd = Math.floor(h / 24)
   130→  return `${dd}d ago`
   131→}

   137→function formatNumber(n: number, digits = 1): string {
   138→  if (!isFinite(n)) return '0'
   139→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   140→  return n.toFixed(digits)

   157→/** Filter activities relevant to the HR user. */
   158→function isHrActivity(a: ActivityItem): boolean {
   159→  const mod = (a.module ?? '').toUpperCase()
   160→  const act = (a.action ?? '').toUpperCase()
   161→  const title = (a.title ?? '').toLowerCase()
   162→  const desc = (a.description ?? '').toLowerCase()
   163→  const hrModules = ['PEOPLE', 'WORKFORCE', 'HR']
   164→  const hrActions = ['WORKFORCE', 'TRAINING', 'EMPLOYEE', 'HIRING', 'ONBOARDING', 'DIVERSITY']
   165→  const hrKeywords = ['employee', 'workforce', 'training', 'worker', 'gender', 'diversity', 'hired', 'onboard', 'hr ']
   166→  return (
   167→    hrModules.includes(mod) ||
   168→    hrActions.some(k => act.includes(k)) ||
   169→    hrKeywords.some(k => title.includes(k) || desc.includes(k))
   170→  )
   171→}

   173→/** Filter action-items relevant to HR. */

   176→  const type = (t.type ?? '').toUpperCase()
   177→  const desc = (t.description ?? '').toLowerCase()
   178→  const title = (t.title ?? '').toLowerCase()

   180→    m === 'submissions' ||
   181→    type === 'CORRECTION' ||
   182→    type === 'DRAFT_SUBMISSION' ||
   183→    type === 'BRSR_GAP' ||
   184→    title.includes('workforce') || title.includes('employee') ||
   185→    desc.includes('workforce') || desc.includes('employee') || desc.includes('training')
   186→  )

   380→/** Derive a deterministic workforce roster from KPI totals. */
   381→function deriveRoster(k: Kpis): {
   382→  code: string; name: string; category: 'Employee' | 'Worker';
   383→  permanent: 'Permanent' | 'Non-Permanent'; gender: 'Male' | 'Female' | 'Other';
   384→  pwd: boolean; department: string
   385→}[] {
   386→  const firstNames = ['Aarav', 'Priya', 'Rohan', 'Ananya', 'Vikram', 'Meera', 'Arjun', 'Isha', 'Karan', 'Sneha',
   387→    'Nikhil', 'Pooja', 'Rahul', 'Divya', 'Aditya', 'Kavya', 'Sahil', 'Riya', 'Manish', 'Tanvi']
   388→  const lastNames = ['Sharma', 'Verma', 'Iyer', 'Nair', 'Reddy', 'Patel', 'Singh', 'Gupta', 'Rao', 'Joshi']
   389→  const departments = ['Operations', 'Finance', 'Engineering', 'HR', 'Safety', 'Logistics', 'Maintenance', 'Administration']
   390→  const seed = (k.totalWorkforce + k.trainingHours) || 1
   391→  const total = Math.min(12, Math.max(8, Math.floor(k.totalWorkforce / 50) || 10))
   392→  const roster: ReturnType<typeof deriveRoster> = []
   393→  for (let i = 0; i < total; i++) {
   394→    const r = ((seed * (i + 7)) % 997) / 997
   395→    const r2 = ((seed * (i + 13)) % 991) / 991
   396→    const r3 = ((seed * (i + 29)) % 977) / 977
   397→    const r4 = ((seed * (i + 41)) % 967) / 967
   398→    const isEmployee = r < 0.55 || i < Math.ceil(total * 0.55)
   399→    const fn = firstNames[(i * 3 + Math.floor(r2 * 5)) % firstNames.length]
   400→    const ln = lastNames[(i * 2 + Math.floor(r3 * 5)) % lastNames.length]
   401→    const gender: 'Male' | 'Female' | 'Other' =
   402→      r2 < k.femaleShare / 100 ? 'Female' : r2 < 0.96 ? 'Male' : 'Other'
   403→    const pwd = r4 < (k.differentlyAbled / Math.max(1, k.totalWorkforce)) * 5 + 0.04
   404→    roster.push({
   405→      code: `MEIL-${(1000 + i).toString()}`,
   406→      name: `${fn} ${ln}`,
   407→      category: isEmployee ? 'Employee' : 'Worker',
   408→      permanent: r3 < 0.78 ? 'Permanent' : 'Non-Permanent',
   409→      gender,
   410→      pwd,
   411→      department: departments[i % departments.length],
   412→    })
   413→  }
   414→  return roster
   415→}

   766→/** Derive a monthly training-hours trend from the trends object + trainingHours. */
   767→function deriveTrainingTrend(trends: Record<string, Record<string, number>> | undefined, k: Kpis): { label: string; hours: number }[] {
   768→  const labels = trends ? Object.keys(trends) : []
   769→  if (labels.length === 0) {
   770→    // synthesize 6 months
   771→    const months = ['Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep']
   772→    const per = k.trainingHours / 6
   773→    return months.map((m, i) => ({
   774→      label: m,
   775→      hours: Math.round(per * (0.7 + (i % 3) * 0.18)),
   776→    }))
   777→  }
   778→  return labels.map((label, i) => {
   779→    const total = k.trainingHours || 0
   780→    const share = 0.55 + ((i * 7) % 9) * 0.05
   781→    return { label, hours: Math.round(total * share / Math.max(1, labels.length)) }
   782→  })
   783→}

  1602→export function HrWorkspace() {
  1603→  const { activeModule } = useApp()
  1604→  const [overview, setOverview] = useState<OverviewData | null>(null)
  1605→  const [activities, setActivities] = useState<ActivityItem[]>([])
  1606→  const [tasks, setTasks] = useState<ActionItem[]>([])
  1607→  const [loading, setLoading] = useState(true)
  1608→  const [error, setError] = useState('')
  1609→  const mountedRef = useRef(true)
  1610→
  1611→  /* ---- fetchers ---- */
  1612→  const fetchOverview = useCallback(async () => {
  1613→    try {
  1614→      const res = await fetch('/api/overview', { cache: 'no-store' })
  1615→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1616→      const data = (await res.json()) as OverviewData
  1617→      if (!mountedRef.current) return
  1618→      setOverview(data)
  1619→      setError('')
  1620→    } catch (e) {
  1621→      if (!mountedRef.current) return
  1622→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
  1623→    }
  1624→  }, [overview])
  1625→
  1626→  const fetchActivities = useCallback(async () => {
  1627→    try {
  1628→      const res = await fetch('/api/activity?take=10', { cache: 'no-store' })
  1629→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1630→      const data = (await res.json()) as ActivityResponse
  1631→      if (!mountedRef.current) return
  1632→      const filtered = (Array.isArray(data.items) ? data.items : []).filter(isHrActivity).slice(0, 5)
  1633→      setActivities(filtered)
  1634→    } catch {
  1635→      /* silent — keep existing feed on poll error */
  1636→    }
  1637→  }, [])
  1638→
  1639→  const fetchTasks = useCallback(async () => {
  1640→    try {
  1641→      const res = await fetch('/api/action-items', { cache: 'no-store' })

  1643→      const data = (await res.json()) as ActionItemsResponse
  1644→      if (!mountedRef.current) return
  1645→      const filtered = (Array.isArray(data.tasks) ? data.tasks : []).filter(isHrTask).slice(0, 6)
  1646→      setTasks(filtered)
  1647→    } catch {
  1648→      /* silent */
  1649→    }
  1650→  }, [])
  1651→
  1652→  /* ---- initial load ---- */
  1653→  useEffect(() => {
  1654→    mountedRef.current = true
  1655→    ;(async () => {
  1656→      setLoading(true)
  1657→      await Promise.all([fetchOverview(), fetchActivities(), fetchTasks()])
  1658→      if (mountedRef.current) setLoading(false)
  1659→    })()
  1660→    return () => { mountedRef.current = false }
  1661→  }, [])
  1662→
  1663→  /* ---- polling: activity every 30s, overview & tasks every 60s ---- */
  1664→  useEffect(() => {
  1665→    const activityTimer = setInterval(fetchActivities, 30_000)
  1666→    const overviewTimer = setInterval(fetchOverview, 60_000)

  1669→      clearInterval(activityTimer)
  1670→      clearInterval(overviewTimer)
  1671→      clearInterval(tasksTimer)
  1672→    }
  1673→  }, [fetchActivities, fetchOverview, fetchTasks])
  1674→
  1675→  const k = useMemo<Kpis | null>(() => {
  1676→    if (!overview?.kpis) return null
  1677→    return overview.kpis
  1678→  }, [overview])
  1679→
  1680→  const trends = useMemo<Record<string, Record<string, number>> | undefined>(() => {
  1681→    const t = overview?.trends as Record<string, Record<string, number>> | undefined
  1682→    return t && typeof t === 'object' ? t : undefined
  1683→  }, [overview])
  1684→
  1685→  const periods = overview?.periods
  1686→
  1687→  // ---- Loading skeleton ----
  1688→  if (loading && !overview) {
  1689→    return <WorkspaceSkeleton tiles={activeModule === 'hr-workforce' ? 4 : 3} />
  1690→  }
  1691→
  1692→  // ---- Error state ----
  1693→  if (error && !overview) {
  1694→    return <ErrorState error={error} onRetry={() => window.location.reload()} />
  1695→  }
  1696→
  1697→  // ---- Empty state ----
  1698→  if (!overview || !k) {
  1699→    return (
  1700→      <EmptyState
  1701→        icon={Users}
  1702→        title="No workforce data yet"
  1703→        subtitle="Set up a reporting period to populate the HR workspace."
  1704→      />
  1705→    )
  1706→  }
  1707→
  1708→  // ---- Dispatch by module ----
  1709→  return (
  1710→    <AnimatePresence mode="wait">
  1711→      <motion.div
  1712→        key={activeModule}
  1713→        initial={{ opacity: 0, y: 8 }}
  1714→        animate={{ opacity: 1, y: 0 }}

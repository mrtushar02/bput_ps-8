    52→/* -------------------------------------------------------- grade + score */
    53→
    54→function gradeFromScore(score: number): { grade: string; label: string } {
    55→  if (score >= 90) return { grade: 'A+', label: 'Exemplary' }
    56→  if (score >= 80) return { grade: 'A',  label: 'Strong' }
    57→  if (score >= 70) return { grade: 'B+', label: 'On Track' }
    58→  if (score >= 60) return { grade: 'B',  label: 'Improving' }
    59→  if (score >= 50) return { grade: 'C',  label: 'Watch' }
    60→  return { grade: 'D', label: 'At Risk' }
    61→}
    62→
    63→/* ------------------------------------------------------- trend helpers */
    64→
    65→function trendDelta(
    66→  trends: { emissions: number; energy: number; water: number; waste: number }[] | undefined,
    67→  key: 'emissions' | 'energy' | 'water' | 'waste'
    68→): number | null {
    69→  if (!trends || trends.length < 2) return null
    70→  const arr = Object.values(trends)
    71→  const last = arr[arr.length - 1][key]
    72→  const prev = arr[arr.length - 2][key]
    73→  if (prev === 0) return null
    74→  return Math.round(((last - prev) / prev) * 1000) / 10
    75→}
    76→
    77→type TrendPill =
    78→  | { kind: 'delta'; value: number; goodWhen: 'up' | 'down' }
    79→  | { kind: 'status'; text: string; tone: 'good' | 'watch' | 'risk' | 'neutral' }

   414→function EsgGauge({ score, grade }: { score: number; grade: string }) {
   415→  const size = 120
   416→  const stroke = 10
   417→  const r = (size - stroke) / 2
   418→  const c = 2 * Math.PI * r
   419→  const pct = Math.max(0, Math.min(100, score)) / 100
   420→  const dash = c * pct
   421→  const gid = 'esg-gauge-amber'
   422→
   423→  return (
   424→    <div className="relative flex-shrink-0" style={{ width: size, height: size }}>
   425→      <svg width={size} height={size} className="-rotate-90">
   426→        <defs>
   427→          <linearGradient id={gid} x1="0%" y1="0%" x2="100%" y2="100%">
   428→            <stop offset="0%" stopColor={AMBER_LITE} />
   429→            <stop offset="55%" stopColor={AMBER} />
   430→            <stop offset="100%" stopColor={GOLD} />
   431→          </linearGradient>
   432→        </defs>
   433→        {/* track */}
   434→        <circle
   435→          cx={size / 2} cy={size / 2} r={r}
   436→          fill="none"
   437→          stroke="rgba(180,83,9,0.14)"
   438→          strokeWidth={stroke}
   439→        />
   440→        {/* progress */}
   441→        <motion.circle
   442→          cx={size / 2} cy={size / 2} r={r}
   443→          fill="none"
   444→          stroke={`url(#${gid})`}
   445→          strokeWidth={stroke}
   446→          strokeLinecap="round"
   447→          strokeDasharray={`${dash} ${c - dash}`}
   448→          initial={{ strokeDasharray: `0 ${c}` }}
   449→          animate={{ strokeDasharray: `${dash} ${c - dash}` }}
   450→          transition={{ duration: 1.1, ease: [0.16, 1, 0.3, 1], delay: 0.25 }}
   451→          style={{ filter: 'drop-shadow(0 2px 6px rgba(245,158,11,0.45))' }}
   452→        />
   453→      </svg>
   454→      {/* center overlay */}
   455→      <div className="pointer-events-none absolute inset-0 flex flex-col items-center justify-center">
   456→        <motion.span
   457→          initial={{ opacity: 0, scale: 0.6 }}
   458→          animate={{ opacity: 1, scale: 1 }}
   459→          transition={{ delay: 0.55, duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   460→          className="text-[28px] font-black leading-none text-amber-950"
   461→        >
   462→          {score}
   463→        </motion.span>
   464→        <motion.span
   465→          initial={{ opacity: 0, y: 2 }}
   466→          animate={{ opacity: 1, y: 0 }}
   467→          transition={{ delay: 0.7, duration: 0.35 }}
   468→          className="mt-0.5 flex h-5 w-5 items-center justify-center rounded-full bg-gradient-to-br from-amber-400 to-amber-600 text-[10px] font-black text-white shadow-sm"
   469→        >
   470→          {grade}
   471→        </motion.span>
   472→        <span className="mt-1 text-[8px] font-semibold uppercase tracking-wider text-amber-700/70">/ 100</span>
   473→      </div>
   474→    </div>
   475→  )
   476→}

   478→/* ====================================================== Trend Pill View */
   479→
   480→function TrendPillView({ pill }: { pill: TrendPill }) {
   481→  if (pill.kind === 'delta') {
   482→    const up = pill.value > 0
   483→    const down = pill.value < 0
   484→    const isGood =
   485→      (pill.goodWhen === 'up' && up) || (pill.goodWhen === 'down' && down)
   486→    const isBad =
   487→      (pill.goodWhen === 'up' && down) || (pill.goodWhen === 'down' && up)
   488→    const toneClass = isGood
   489→      ? 'bg-emerald-100 text-emerald-700 border-emerald-200/60'
   490→      : isBad
   491→        ? 'bg-rose-100 text-rose-700 border-rose-200/60'
   492→        : 'bg-amber-100 text-amber-800 border-amber-200/60'
   493→    const Icon = up ? TrendingUp : down ? TrendingDown : Activity
   494→    const text = pill.value === 0 ? '0%' : `${up ? '+' : ''}${pill.value}%`
   495→    return (
   496→      <span className={`inline-flex items-center gap-0.5 rounded-full border px-1.5 py-0.5 text-[9px] font-bold ${toneClass}`}>
   497→        <Icon className="h-2.5 w-2.5" strokeWidth={2.5} />
   498→        {text}
   499→      </span>
   500→    )
   501→  }
   502→  // status pill
   503→  const map: Record<string, string> = {
   504→    good:    'bg-emerald-100 text-emerald-700 border-emerald-200/60',
   505→    watch:   'bg-amber-100 text-amber-800 border-amber-200/60',
   506→    risk:    'bg-rose-100 text-rose-700 border-rose-200/60',
   507→    neutral: 'bg-amber-50 text-amber-700/80 border-amber-200/50',
   508→  }
   509→  return (
   510→    <span className={`inline-flex items-center rounded-full border px-1.5 py-0.5 text-[9px] font-bold ${map[pill.tone]}`}>
   511→      {pill.text}
   512→    </span>
   513→  )
   514→}

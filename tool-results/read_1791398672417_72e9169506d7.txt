   122→function timeAgo(iso: string): string {
   123→  const d = new Date(iso)
   124→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   125→  if (s < 30) return 'just now'
   126→  if (s < 60) return `${s}s ago`
   127→  const m = Math.floor(s / 60)
   128→  if (m < 60) return `${m}m ago`
   129→  const h = Math.floor(m / 60)
   130→  if (h < 24) return `${h}h ago`
   131→  const dd = Math.floor(h / 24)
   132→  return `${dd}d ago`
   133→}

   139→function formatNumber(n: number, digits = 1): string {
   140→  if (!isFinite(n)) return '0'
   141→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   142→  return n.toFixed(digits)
   143→}

   159→/** Filter activities to EHS / safety / incident actions. */
   160→function isSafetyActivity(a: ActivityItem): boolean {
   161→  const mod = (a.module ?? '').toUpperCase()
   162→  const act = (a.action ?? '').toUpperCase()
   163→  const title = (a.title ?? '').toLowerCase()
   164→  const desc = (a.description ?? '').toLowerCase()
   165→  const safetyModules = ['SAFETY', 'EHS']
   166→  const safetyActions = ['INCIDENT', 'INJURY', 'LTI', 'FATALITY', 'SAFETY', 'CORRECTIVE', 'TRAINING_SAFETY']
   167→  const safetyKeywords = ['incident', 'injury', 'safety', 'lti', 'fatality', 'corrective', 'ppe', 'hazard', 'near miss', 'fire drill']
   168→  return (
   169→    safetyModules.includes(mod) ||
   170→    safetyActions.some(k => act.includes(k)) ||
   171→    safetyKeywords.some(k => title.includes(k) || desc.includes(k))
   172→  )
   173→}

   175→/** Filter action-items relevant to the EHS / safety user role. */
   176→function isSafetyTask(t: ActionItem): boolean {
   177→  const type = (t.type ?? '').toUpperCase()
   178→  const title = (t.title ?? '').toLowerCase()
   179→  const desc = (t.description ?? '').toLowerCase()
   180→  return (
   181→    type === 'CORRECTION' ||
   182→    type === 'VALIDATION_ERROR' ||
   183→    title.includes('safety') || title.includes('incident') || title.includes('injury') ||
   184→    desc.includes('safety') || desc.includes('incident') || desc.includes('injury') || desc.includes('corrective')
   185→  )
   186→}

   188→/** Build a monthly incident trend from totals + periods. */
   189→function buildIncidentTrend(
   190→  periods: { id: string; label: string }[] | undefined,
   191→  k: Kpis,
   192→): { label: string; injuries: number; lti: number; fatalities: number }[] {
   193→  const totalInjuries = Math.max(k.injuries, 0)
   194→  const totalLti = Math.max(k.lti, 0)
   195→  const totalFatal = Math.max(k.fatalities, 0)
   196→
   197→  // Use real period labels when available; otherwise synthesize a 6-month rolling window.
   198→  let labels: string[]
   199→  if (periods && periods.length > 0) {
   200→    labels = periods.slice(-6).map(p => p.label)
   201→  } else {
   202→    const now = new Date()
   203→    labels = Array.from({ length: 6 }).map((_, i) => {
   204→      const d = new Date(now.getFullYear(), now.getMonth() - (5 - i), 1)
   205→      return d.toLocaleString(undefined, { month: 'short' })
   206→    })
   207→  }
   208→
   209→  // Distribute the totals deterministically across the periods — latest period
   210→  // holds the bulk of recorded incidents, earlier periods decay toward zero.
   211→  // This is a derived trend visualization; the absolute totals match the KPIs.
   212→  const weights = [0.05, 0.08, 0.12, 0.15, 0.20, 0.40]
   213→  return labels.map((label, i) => {
   214→    const w = weights[i] ?? weights[weights.length - 1]
   215→    return {
   216→      label,
   217→      injuries: Math.round(totalInjuries * w),
   218→      lti: Math.round(totalLti * w),
   219→      fatalities: i === labels.length - 1 ? totalFatal : 0,
   220→    }
   221→  })
   222→}

   809→      const res = await fetch('/api/activity?take=10', { cache: 'no-store' })

   816→      /* silent */
   817→    }

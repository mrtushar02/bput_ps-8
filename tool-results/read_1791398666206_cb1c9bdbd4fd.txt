   120→   120→function timeAgo(iso: string): string {
   121→   121→  const d = new Date(iso)
   122→   122→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   123→   123→  if (s < 30) return 'just now'
   124→   124→  if (s < 60) return `${s}s ago`
   125→   125→  const m = Math.floor(s / 60)
   126→   126→  if (m < 60) return `${m}m ago`
   127→   127→  const h = Math.floor(m / 60)
   128→   128→  if (h < 24) return `${h}h ago`
   129→   129→  const dd = Math.floor(h / 24)
   130→   130→  return `${dd}d ago`
   131→   131→}

   137→   137→function formatNumber(n: number, digits = 1): string {
   138→   138→  if (!isFinite(n)) return '0'
   139→   139→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   140→   140→  return n.toFixed(digits)
   141→   141→}

   143→   143→function statusClass(status?: string | null): string {

   158→   158→function isHrActivity(a: ActivityItem): boolean {
   159→   159→  const mod = (a.module ?? '').toUpperCase()
   160→   160→  const act = (a.action ?? '').toUpperCase()
   161→   161→  const title = (a.title ?? '').toLowerCase()
   162→   162→  const desc = (a.description ?? '').toLowerCase()
   163→   163→  const hrModules = ['PEOPLE', 'WORKFORCE', 'HR']
   164→   164→  const hrActions = ['WORKFORCE', 'TRAINING', 'EMPLOYEE', 'HIRING', 'ONBOARDING', 'DIVERSITY']
   165→   165→  const hrKeywords = ['employee', 'workforce', 'training', 'worker', 'gender', 'diversity', 'hired', 'onboard', 'hr ']
   166→   166→  return (
   167→   167→    hrModules.includes(mod) ||
   168→   168→    hrActions.some(k => act.includes(k)) ||
   169→   169→    hrKeywords.some(k => title.includes(k) || desc.includes(k))
   170→   170→  )

   380→   380→/** Derive a deterministic workforce roster from KPI totals. */
   381→   381→function deriveRoster(k: Kpis): {
   382→   382→  code: string; name: string; category: 'Employee' | 'Worker';
   383→   383→  permanent: 'Permanent' | 'Non-Permanent'; gender: 'Male' | 'Female' | 'Other';
   384→   384→  pwd: boolean; department: string
   385→   385→}[] {
   386→   386→  const firstNames = ['Aarav', 'Priya', 'Rohan', 'Ananya', 'Vikram', 'Meera', 'Arjun', 'Isha', 'Karan', 'Sneha',
   387→   387→    'Nikhil', 'Pooja', 'Rahul', 'Divya', 'Aditya', 'Kavya', 'Sahil', 'Riya', 'Manish', 'Tanvi']
   388→   388→  const lastNames = ['Sharma', 'Verma', 'Iyer', 'Nair', 'Reddy', 'Patel', 'Singh', 'Gupta', 'Rao', 'Joshi']
   389→   389→  const departments = ['Operations', 'Finance', 'Engineering', 'HR', 'Safety', 'Logistics', 'Maintenance', 'Administration']
   390→   390→  const seed = (k.totalWorkforce + k.trainingHours) || 1
   391→   391→  const total = Math.min(12, Math.max(8, Math.floor(k.totalWorkforce / 50) || 10))
   392→   392→  const roster: ReturnType<typeof deriveRoster> = []
   393→   393→  for (let i = 0; i < total; i++) {
   394→   394→    const r = ((seed * (i + 7)) % 997) / 997
   395→   395→    const r2 = ((seed * (i + 13)) % 991) / 991
   396→   396→    const r3 = ((seed * (i + 29)) % 977) / 977
   397→   397→    const r4 = ((seed * (i + 41)) % 967) / 967
   398→   398→    const isEmployee = r < 0.55 || i < Math.ceil(total * 0.55)
   399→   399→    const fn = firstNames[(i * 3 + Math.floor(r2 * 5)) % firstNames.length]
   400→   400→    const ln = lastNames[(i * 2 + Math.floor(r3 * 5)) % lastNames.length]
   401→   401→    const gender: 'Male' | 'Female' | 'Other' =
   402→   402→      r2 < k.femaleShare / 100 ? 'Female' : r2 < 0.96 ? 'Male' : 'Other'
   403→   403→    const pwd = r4 < (k.differentlyAbled / Math.max(1, k.totalWorkforce)) * 5 + 0.04
   404→   404→    roster.push({
   405→   405→      code: `MEIL-${(1000 + i).toString()}`,
   406→   406→      name: `${fn} ${ln}`,
   407→   407→      category: isEmployee ? 'Employee' : 'Worker',
   408→   408→      permanent: r3 < 0.78 ? 'Permanent' : 'Non-Permanent',
   409→   409→      gender,
   410→   410→      pwd,
   411→   411→      department: departments[i % departments.length],
   412→   412→    })
   413→   413→  }
   414→   414→  return roster

   766→   766→/** Derive a monthly training-hours trend from the trends object + trainingHours. */
   767→   767→function deriveTrainingTrend(trends: Record<string, Record<string, number>> | undefined, k: Kpis): { label: string; hours: number }[] {
   768→   768→  const labels = trends ? Object.keys(trends) : []
   769→   769→  if (labels.length === 0) {
   770→   770→    // synthesize 6 months
   771→   771→    const months = ['Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep']
   772→   772→    const per = k.trainingHours / 6
   773→   773→    return months.map((m, i) => ({
   774→   774→      label: m,
   775→   775→      hours: Math.round(per * (0.7 + (i % 3) * 0.18)),
   776→   776→    }))
   777→   777→  }
   778→   778→  return labels.map((label, i) => {
   779→   779→    const total = k.trainingHours || 0
   780→   780→    const share = 0.55 + ((i * 7) % 9) * 0.05
   781→   781→    return { label, hours: Math.round(total * share / Math.max(1, labels.length)) }
   782→   782→  })
   783→   783→}

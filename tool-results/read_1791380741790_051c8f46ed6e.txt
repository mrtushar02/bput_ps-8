   170→   170→export function SiteUserOverview() {
   171→   171→  const { setActiveModule } = useApp()
   172→   172→  const [overview, setOverview] = useState<OverviewData | null>(null)
   173→   173→  const [activities, setActivities] = useState<ActivityItem[]>([])
   174→   174→  const [submissions, setSubmissions] = useState<SubmissionItem[]>([])
   175→   175→  const [loading, setLoading] = useState(true)
   176→   176→  const [error, setError] = useState('')
   177→   177→  const [lastActivityAt, setLastActivityAt] = useState<Date | null>(null)
   178→   178→  const mountedRef = useRef(true)
   179→   179→
   180→   180→  /* ---- fetchers ---- */
   181→   181→  const fetchOverview = useCallback(async () => {
   182→   182→    try {
   183→   183→      const res = await fetch('/api/overview', { cache: 'no-store' })
   184→   184→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   185→   185→      const data = (await res.json()) as OverviewData
   186→   186→      if (!mountedRef.current) return
   187→   187→      setOverview(data)
   188→   188→      setError('')
   189→   189→    } catch (e) {
   190→   190→      if (!mountedRef.current) return
   191→   191→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
   192→   192→    }
   193→   193→  }, [overview])
   194→   194→
   195→   195→  const fetchActivities = useCallback(async () => {
   196→   196→    try {
   197→   197→      const res = await fetch('/api/activity?take=20', { cache: 'no-store' })
   198→   198→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   199→   199→      const data = (await res.json()) as ActivityResponse
   200→   200→      if (!mountedRef.current) return
   201→   201→      setActivities(Array.isArray(data.items) ? data.items : [])
   202→   202→      setLastActivityAt(new Date())
   203→   203→    } catch {
   204→   204→      /* silent — keep existing feed on poll error */
   205→   205→    }
   206→   206→  }, [])
   207→   207→
   208→   208→  const fetchSubmissions = useCallback(async () => {
   209→   209→    try {
   210→   210→      const res = await fetch('/api/submissions?take=10', { cache: 'no-store' })
   211→   211→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   212→   212→      const data = (await res.json()) as SubmissionResponse
   213→   213→      if (!mountedRef.current) return
   214→   214→      setSubmissions(Array.isArray(data.items) ? data.items : [])
   215→   215→    } catch {
   216→   216→      /* silent */
   217→   217→    }
   218→   218→  }, [])
   219→   219→
   220→   220→  /* ---- initial load ---- */
   221→   221→  useEffect(() => {
   222→   222→    mountedRef.current = true
   223→   223→    ;(async () => {
   224→   224→      setLoading(true)
   225→   225→      await Promise.all([fetchOverview(), fetchActivities(), fetchSubmissions()])
   226→   226→      if (mountedRef.current) setLoading(false)
   227→   227→    })()
   228→   228→    return () => { mountedRef.current = false }
   229→   229→  }, [])
   230→   230→
   231→   231→  /* ---- polling: activity every 30s, overview every 60s ---- */
   232→   232→  useEffect(() => {
   233→   233→    const activityTimer = setInterval(fetchActivities, 30_000)
   234→   234→    const overviewTimer = setInterval(fetchOverview, 60_000)
   235→   235→    return () => {
   236→   236→      clearInterval(activityTimer)
   237→   237→      clearInterval(overviewTimer)
   238→   238→    }
   239→   239→  }, [fetchActivities, fetchOverview])
   240→   240→
   241→   241→  /* ---- derived data ---- */
   242→   242→  const trendArr = useMemo<Array<Trend & { label: string }>>(() => {
   243→   243→    if (!overview) return []
   244→   244→    return Object.entries(overview.trends).map(([label, v]) => ({ label, ...v }))
   245→   245→  }, [overview])
   246→   246→
   247→   247→  const emissionsBySourceArr = useMemo(() => {
   248→   248→    if (!overview) return [] as { name: string; value: number }[]
   249→   249→    return Object.entries(overview.emissionsBySource)
   250→   250→      .map(([name, value]) => ({ name, value: Math.round(value * 100) / 100 }))
   251→   251→      .filter(d => d.value > 0)
   252→   252→      .sort((a, b) => b.value - a.value)
   253→   253→  }, [overview])
   254→   254→
   255→   255→  const monthlyEnergy = useMemo(() => {
   256→   256→    return trendArr.map(t => ({ label: shortMonth(t.label), value: Math.round(t.energy) }))
   257→   257→  }, [trendArr])
   258→   258→
   259→   259→  const moduleCompletion = useMemo(() => {
   260→   260→    const MODULES = ['ENERGY', 'WATER', 'WASTE', 'SAFETY', 'PEOPLE'] as const
   261→   261→    const fallback: Record<string, number> = { ENERGY: 0, WATER: 0, WASTE: 0, SAFETY: 0, PEOPLE: 0 }
   262→   262→    if (!submissions.length) {
   263→   263→      // If no submissions, derive a soft % from the KPIs so the bar isn't empty
   264→   264→      const k = overview?.kpis
   265→   265→      if (!k) return []
   266→   266→      return [
   267→   267→        { module: 'Energy',   pct: Math.min(100, Math.round((k.energyGJ > 0 ? 70 : 30) + (k.renewableShare / 5))) },
   268→   268→        { module: 'Water',    pct: Math.min(100, Math.round((k.waterWithdrawalKL > 0 ? 65 : 25) + (k.waterRecycledShare / 5))) },
   269→   269→        { module: 'Waste',    pct: Math.min(100, Math.round((k.wasteGeneratedT > 0 ? 60 : 20) + (k.wasteRecycledShare / 5))) },
   270→   270→        { module: 'Safety',   pct: Math.min(100, Math.round(k.totalWorkforce > 0 ? 80 : 10)) },
   271→   271→        { module: 'Workforce', pct: Math.min(100, Math.round(k.totalWorkforce > 0 ? 75 : 10)) },
   272→   272→      ]
   273→   273→    }
   274→   274→    const byMod: Record<string, SubmissionItem[]> = {}
   275→   275→    for (const s of submissions) {
   276→   276→      const m = (s.module || '').toUpperCase()
   277→   277→      if (!MODULES.includes(m as typeof MODULES[number])) continue
   278→   278→      ;(byMod[m] ??= []).push(s)
   279→   279→    }
   280→   280→    const moduleLabels: Record<string, string> = { ENERGY: 'Energy', WATER: 'Water', WASTE: 'Waste', SAFETY: 'Safety', PEOPLE: 'Workforce' }
   281→   281→    return MODULES.map(m => {
   282→   282→      const items = byMod[m] ?? []
   283→   283→      const pct = items.length === 0
   284→   284→        ? fallback[m]
   285→   285→        : Math.round(items.reduce((s, x) => s + (x.completionPct || 0), 0) / items.length)
   286→   286→      return { module: moduleLabels[m], pct: Math.min(100, Math.max(0, pct)) }
   287→   287→    })
   288→   288→  }, [submissions, overview])
   289→   289→
   290→   290→  const activeSubs = useMemo(() => submissions.slice(0, 5), [submissions])
   291→   291→
   292→   292→  /* ---- render states ---- */
   293→   293→  if (loading) return <DashboardSkeleton />
   294→   294→  if (error && !overview) return <ErrorState message={error} onRetry={() => location.reload()} />
   295→   295→  if (!overview) return <EmptyState />
   296→   296→  const k = overview.kpis
   297→   297→
   298→   298→  /* ---- KPI cards data (Emissions / Energy / Water) ---- */
   299→   299→  const kpiCards = [
   300→   300→    {
   301→   301→      icon: Flame,
   302→   302→      label: 'Emissions',
   303→   303→      value: k.totalEmissions.toLocaleString(),
   304→   304→      unit: 'tCO₂e',
   305→   305→      trend: trendDelta(trendArr, 'emissions'),
   306→   306→      spark: trendArr.map(t => t.emissions),
   307→   307→      sub: `S1: ${k.scope1.toLocaleString()} · S2: ${k.scope2.toLocaleString()}`,
   308→   308→      goodDirection: 'down' as const,
   309→   309→      tone: 'bg-rose-50 text-rose-600',
   310→   310→      module: 'analytics' as ModuleKey,
   311→   311→    },
   312→   312→    {
   313→   313→      icon: Zap,
   314→   314→      label: 'Energy',
   315→   315→      value: k.energyGJ.toLocaleString(),
   316→   316→      unit: 'GJ',
   317→   317→      trend: trendDelta(trendArr, 'energy'),
   318→   318→      spark: trendArr.map(t => t.energy),
   319→   319→      sub: `Renewable ${k.renewableShare}%`,
   320→   320→      goodDirection: 'down' as const,
   321→   321→      tone: 'bg-amber-50 text-amber-600',
   322→   322→      module: 'data-entry' as ModuleKey,
   323→   323→    },
   324→   324→    {
   325→   325→      icon: Droplet,
   326→   326→      label: 'Water',
   327→   327→      value: k.waterWithdrawalKL.toLocaleString(),
   328→   328→      unit: 'KL',
   329→   329→      trend: trendDelta(trendArr, 'water'),
   330→   330→      spark: trendArr.map(t => t.water),
   331→   331→      sub: `Recycled ${k.waterRecycledShare}%`,
   332→   332→      goodDirection: 'down' as const,
   333→   333→      tone: 'bg-cyan-50 text-cyan-600',
   334→   334→      module: 'data-entry' as ModuleKey,
   335→   335→    },
   336→   336→  ]
   337→   337→
   338→   338→  return (
   339→   339→    <div className="space-y-5">
   340→   340→      {/* ---- Page header ---- */}
   341→   341→      <PageHeader k={k} periodCount={overview.periods.length} />
   342→   342→
   343→   343→      {/* ---- 3-column asymmetric grid ---- */}
   344→   344→      <div className="grid grid-cols-1 gap-5 xl:grid-cols-[1fr_400px_280px]">
   345→   345→        {/* ============================
   346→   346→            LEFT COLUMN (~58%)
   347→   347→           ============================ */}
   348→   348→        <div className="space-y-5">
   349→   349→          {/* KPI cards row */}
   350→   350→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   351→   351→            {kpiCards.map((c, i) => (
   352→   352→              <KpiCard key={c.label} delay={0.05 * i} {...c} onClick={() => setActiveModule(c.module)} />
   353→   353→            ))}
   354→   354→          </div>
   355→   355→
   356→   356→          {/* Active Submissions table */}
   357→   357→          <ActiveSubmissionsCard
   358→   358→            subs={activeSubs}
   359→   359→            onViewAll={() => setActiveModule('submissions')}
   360→   360→          />
   361→   361→
   362→   362→          {/* Data Entry Status bar */}
   363→   363→          <DataEntryStatusCard modules={moduleCompletion} onOpen={() => setActiveModule('data-entry')} />
   364→   364→        </div>
   365→   365→
   366→   366→        {/* ============================
   367→   367→            CENTER COLUMN (~25%)
   368→   368→           ============================ */}
   369→   369→        <div className="space-y-5">
   370→   370→          <RecentActivitiesCard
   371→   371→            activities={activities}
   372→   372→            lastActivityAt={lastActivityAt}
   373→   373→          />
   374→   374→          <TeamSubmissionsCard
   375→   375→            team={SEEDED_TEAM}
   376→   376→            activeCount={SEEDED_TEAM.filter(t => t.active).length}
   377→   377→            onOpen={() => setActiveModule('my-project')}
   378→   378→          />
   379→   379→        </div>
   380→   380→
   381→   381→        {/* ============================
   382→   382→            RIGHT COLUMN (~17%)
   383→   383→           ============================ */}
   384→   384→        <div className="space-y-5">
   385→   385→          <AnalyticsMiniCard
   386→   386→            emissionsBySource={emissionsBySourceArr}
   387→   387→            monthlyEnergy={monthlyEnergy}
   388→   388→            onOpen={() => setActiveModule('analytics')}

  1125→  1125→function trendDelta(trends: Trend[], key: keyof Trend): number | null {
  1126→  1126→  if (!trends || trends.length < 2) return null
  1127→  1127→  const last = trends[trends.length - 1][key]
  1128→  1128→  const prev = trends[trends.length - 2][key]
  1129→  1129→  if (prev === 0) return null
  1130→  1130→  return Math.round(((last - prev) / prev) * 1000) / 10
  1131→  1131→}

     1→'use client'
     2→/**
     3→ * Data Entry Module — flagship 4-step workflow screen.
     4→ * Steps: 1 Enter Data → 2 Attach Evidence → 3 Validate → 4 Submit.
     5→ * Sub-modules: energy | water | waste | workforce | safety | travel.
     6→ * Real data via /api/{module} POST + /api/validation/run + /api/submissions + /api/submissions/[id]/submit.
     7→ * No hardcoded KPIs. Role-aware: reviewer roles get read-only forms.
     8→ */
     9→import { useEffect, useState, useCallback, useRef } from 'react'
    10→import { motion, AnimatePresence } from 'framer-motion'
    11→import {
    12→  FileText, Zap, Droplet, Recycle, Users, ShieldCheck, Plane, CheckCircle2,
    13→  AlertTriangle, AlertOctagon, Clock, Link2, Send, Save, FlaskConical,
    14→  ChevronRight, ChevronLeft, Building2, CalendarClock, Info, Lock, Calculator, type LucideIcon,
    15→} from 'lucide-react'
    16→import { useApp } from '@/lib/auth-context'
    17→
    18→/* ---------- Types ---------- */
    19→type SubModule = 'energy' | 'water' | 'waste' | 'workforce' | 'safety' | 'travel'
    20→
    21→interface Project {
    22→  id: string; projectCode: string; projectName: string; location: string | null
    23→  buName: string; subsidiaryCode: string; groupCode: string
    24→}
    25→interface Period { id: string; label: string; year: number; month: number | null; status: string }
    26→interface EvidenceItem {
    27→  id: string; fileName: string; documentType: string; status: string
    28→  module?: string | null
    29→}
    30→interface ValidationIssue {
    31→  ruleCode: string; severity: 'INFO' | 'WARNING' | 'ERROR' | 'BLOCKING'
    32→  message: string; field?: string | null; suggestedAction?: string | null
    33→}
    34→interface CalcPayload {
    35→  calculatedValue: number; resultUnit: string; scope?: string | null
    36→  factorId?: string | null; factorVersion?: number | null
    37→  methodologyNote?: string | null
    38→  normalizedValue?: number | null; normalizedUnit?: string | null
    39→  sourceValue?: number; sourceUnit?: string
    40→}
    41→interface SavedRecord {
    42→  id: string
    43→  recordType: string
    44→  issues: ValidationIssue[]
    45→  calculation: CalcPayload | null
    46→  derivedLtifr?: number | null
    47→  validationStatus?: string
    48→  evidenceId?: string | null
    49→}
    50→
    51→/* ---------- Role helpers ---------- */
    52→const READ_ONLY_ROLES = new Set([
    53→  'BU_REVIEWER', 'SUBSIDIARY_REVIEWER', 'GROUP_REVIEWER', 'AUDITOR', 'EXECUTIVE',
    54→])
    55→
    56→const SUB_MODULES: Array<{ key: SubModule; label: string; icon: LucideIcon; moduleKey: string; recordType: string; permission: string }> = [
    57→  { key: 'energy', label: 'Energy / Fuel', icon: Zap, moduleKey: 'ENERGY', recordType: 'ENERGY', permission: 'esg.energy.write' },
    58→  { key: 'water', label: 'Water', icon: Droplet, moduleKey: 'WATER', recordType: 'WATER', permission: 'esg.water.write' },
    59→  { key: 'waste', label: 'Waste', icon: Recycle, moduleKey: 'WASTE', recordType: 'WASTE', permission: 'esg.waste.write' },
    60→  { key: 'workforce', label: 'Workforce', icon: Users, moduleKey: 'PEOPLE', recordType: 'PEOPLE', permission: 'esg.people.write' },
    61→  { key: 'safety', label: 'Safety', icon: ShieldCheck, moduleKey: 'SAFETY', recordType: 'SAFETY', permission: 'esg.safety.write' },
    62→  { key: 'travel', label: 'Travel', icon: Plane, moduleKey: 'TRAVEL', recordType: 'TRAVEL', permission: 'esg.travel.write' },
    63→]
    64→
    65→const STEPS = ['Enter Data', 'Attach Evidence', 'Validate', 'Submit'] as const
    66→
    67→/* ---------- Main component ---------- */
    68→export function DataEntryModule({ subModule: subModuleProp }: { subModule: string }) {
    69→  const { user, setDataEntrySubModule } = useApp()
    70→  const roleKey = user?.roles?.[0]?.key ?? ''
    71→  const readOnly = READ_ONLY_ROLES.has(roleKey)
    72→
    73→  const subModule = (SUB_MODULES.find(s => s.key === subModuleProp)?.key ?? 'energy') as SubModule
    74→  const activeSub = SUB_MODULES.find(s => s.key === subModule) ?? SUB_MODULES[0]
    75→
    76→  const [projects, setProjects] = useState<Project[]>([])
    77→  const [periods, setPeriods] = useState<Period[]>([])
    78→  const [evidence, setEvidence] = useState<EvidenceItem[]>([])
    79→  const [selectedProjectId, setSelectedProjectId] = useState<string | null>(null)
    80→  const [selectedPeriodId, setSelectedPeriodId] = useState<string | null>(null)
    81→  const [loading, setLoading] = useState(true)
    82→  const [error, setError] = useState('')
    83→
    84→  // Stepper / action state
    85→  const [step, setStep] = useState<1 | 2 | 3 | 4>(1)
    86→  const [saved, setSaved] = useState<SavedRecord | null>(null)
    87→  const [submitting, setSubmitting] = useState(false)
    88→  const [actionError, setActionError] = useState('')
    89→  const [actionInfo, setActionInfo] = useState('')
    90→  const [validating, setValidating] = useState(false)
    91→  const [validationRun, setValidationRun] = useState<{ passed: number; errors: number; warnings: number } | null>(null)
    92→  const [submitResult, setSubmitResult] = useState<{ submissionId: string; status: string } | null>(null)
    93→  const [existingRecords, setExistingRecords] = useState<any[]>([])
    94→
    95→  // Form state per submodule (so user doesn't lose data on tab switch)
    96→  const formStateRef = useRef<Record<string, Record<string, any>>>({})
    97→
    98→  // ---- Load projects + periods once
    99→  useEffect(() => {
   100→    let cancelled = false
   101→    setLoading(true); setError('')
   102→    Promise.all([
   103→      fetch('/api/organization/tree').then(r => r.ok ? r.json() : Promise.reject(new Error('org tree failed'))),
   104→      fetch('/api/overview').then(r => r.ok ? r.json() : Promise.reject(new Error('overview failed'))),
   105→    ])
   106→      .then(([tr, ov]) => {
   107→        if (cancelled) return
   108→        const flat: Project[] = []
   109→        for (const g of tr.groups || []) {
   110→          for (const sub of g.subsidiaries || []) {
   111→            for (const bu of sub.businessUnits || []) {
   112→              for (const p of bu.projects || []) {
   113→                flat.push({
   114→                  id: p.id, projectCode: p.projectCode, projectName: p.projectName,
   115→                  location: p.location, buName: bu.name, subsidiaryCode: sub.code, groupCode: g.code,
   116→                })
   117→              }
   118→            }
   119→          }
   120→        }
   121→        setProjects(flat)
   122→        setPeriods(ov.periods || [])
   123→        const scopeProj = user?.scopes?.find(s => s.scopeType === 'PROJECT')
   124→        const defaultProject =
   125→          (scopeProj ? flat.find(p => p.id === scopeProj.scopeId) : null) ||
   126→          flat.find(p => p.projectCode === 'MEIL-SOL-GJT') ||
   127→          flat[0] || null
   128→        setSelectedProjectId(defaultProject?.id ?? null)
   129→        // Default period: prefer June (open/draft) → May → April → latest
   130→        const defaultPeriod =
   131→          (ov.periods || []).find((p: Period) => p.label === 'June 2026') ||
   132→          (ov.periods || []).find((p: Period) => p.label === 'May 2026') ||
   133→          (ov.periods || [])[0] || null
   134→        setSelectedPeriodId(defaultPeriod?.id ?? null)
   135→        setLoading(false)
   136→      })
   137→      .catch((e: any) => { if (!cancelled) { setError(e?.message || 'Failed to load'); setLoading(false) } })
   138→    return () => { cancelled = true }
   139→  }, [user?.scopes])
   140→
   141→  // ---- Reset step + saved record when submodule/project/period changes
   142→  useEffect(() => {
   143→    setStep(1); setSaved(null); setActionError(''); setActionInfo('')
   144→    setValidationRun(null); setSubmitResult(null)
   145→  }, [subModule, selectedProjectId, selectedPeriodId])
   146→
   147→  // ---- Fetch existing records for the project+period+module (so user sees "no records" empty state)
   148→  const refreshExistingRecords = useCallback(async () => {
   149→    if (!selectedProjectId || !selectedPeriodId) { setExistingRecords([]); return }
   150→    try {
   151→      const endpoint = activeSub.key === 'travel' ? null : `/api/${activeSub.key}`
   152→      if (!endpoint) { setExistingRecords([]); return }
   153→      const r = await fetch(`${endpoint}?projectId=${encodeURIComponent(selectedProjectId)}&periodId=${encodeURIComponent(selectedPeriodId)}`)
   154→      if (!r.ok) { setExistingRecords([]); return }
   155→      const d = await r.json()
   156→      setExistingRecords(d.records || [])
   157→    } catch {
   158→      setExistingRecords([])
   159→    }
   160→  }, [selectedProjectId, selectedPeriodId, activeSub.key])
   161→
   162→  useEffect(() => { refreshExistingRecords() }, [refreshExistingRecords])
   163→
   164→  // ---- Fetch evidence for this module + project
   165→  useEffect(() => {
   166→    if (!selectedProjectId) { setEvidence([]); return }
   167→    fetch(`/api/evidence?projectId=${encodeURIComponent(selectedProjectId)}&module=${activeSub.moduleKey}`)
   168→      .then(r => r.ok ? r.json() : { items: [] })
   169→      .then(d => setEvidence(d.items || []))
   170→      .catch(() => setEvidence([]))
   171→  }, [selectedProjectId, activeSub.moduleKey])
   172→
   173→  const selectedProject = projects.find(p => p.id === selectedProjectId) || null
   174→  const selectedPeriod = periods.find(p => p.id === selectedPeriodId) || null
   175→
   176→  if (loading) return <DataEntrySkeleton />
   177→  if (error) return <ErrorState message={error} onRetry={() => location.reload()} />
   178→  if (projects.length === 0 || periods.length === 0) {
   179→    return <EmptyState message="No projects or reporting periods are available in your scope. Contact your administrator." />
   180→  }
   181→
   182→  // ---- Handlers
   183→  const handleSaved = (result: SavedRecord) => {
   184→    setSaved(result)
   185→    setStep(2)
   186→    setActionInfo('Draft saved. Validation issues and calculation preview shown on the right.')
   187→    refreshExistingRecords()
   188→  }
   189→
   190→  const handleAttachEvidence = async (evidenceId: string | null) => {
   191→    if (!saved) return
   192→    setActionError('')
   193→    try {
   194→      setSubmitting(true)
   195→      // Best-effort: re-submit the saved record with evidenceId by re-running POST
   196→      // We just update the local SavedRecord's evidenceId; the actual link is persisted
   197→      // when the user clicks Save Draft again or via the submit flow.
   198→      setSaved(prev => prev ? { ...prev, evidenceId } : prev)
   199→      setStep(3)
   200→    } finally {
   201→      setSubmitting(false)
   202→    }
   203→  }
   204→
   205→  const handleValidate = async () => {
   206→    if (!saved) { setActionError('Save a draft first.'); return }
   207→    setActionError(''); setActionInfo(''); setValidating(true)
   208→    try {
   209→      const r = await fetch('/api/validation/run', {
   210→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   211→        body: JSON.stringify({ recordType: saved.recordType, recordId: saved.id }),
   212→      })
   213→      const d = await r.json()
   214→      if (!r.ok) throw new Error(d.error || 'Validation failed')
   215→      setValidationRun({ passed: d.passed ?? 0, errors: d.errors ?? 0, warnings: d.warnings ?? 0 })
   216→      // refresh issues from the per-record endpoint
   217→      const endpoint = activeSub.key === 'travel' ? null : `/api/${activeSub.key}`
   218→      if (endpoint) {
   219→        const list = await fetch(`${endpoint}?projectId=${selectedProjectId}&periodId=${selectedPeriodId}`).then(rr => rr.json())
   220→        const rec = (list.records || []).find((x: any) => x.id === saved.id)
   221→        if (rec) {
   222→          setSaved(prev => prev ? {
   223→            ...prev,
   224→            issues: (rec.validationResults || []).map((v: any) => ({
   225→              ruleCode: v.ruleCode, severity: v.severity, message: v.message,
   226→              field: v.field, suggestedAction: v.suggestedAction,
   227→            })),
   228→            validationStatus: rec.validationStatus,
   229→          } : prev)
   230→        }
   231→      }
   232→      setStep(4)
   233→    } catch (e: any) {
   234→      setActionError(e?.message || 'Validation failed')
   235→    } finally {
   236→      setValidating(false)
   237→    }
   238→  }
   239→
   240→  const handleSubmit = async () => {
   241→    if (!saved) { setActionError('Save a draft first.'); return }
   242→    if (!selectedProjectId || !selectedPeriodId) { setActionError('Select a project and period.'); return }
   243→    setActionError(''); setActionInfo(''); setSubmitting(true)
   244→    try {
   245→      // 1. Create the submission (DRAFT)
   246→      const createRes = await fetch('/api/submissions', {
   247→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   248→        body: JSON.stringify({
   249→          projectId: selectedProjectId, reportingPeriodId: selectedPeriodId,
   250→          module: activeSub.moduleKey, recordIds: [saved.id],
   251→        }),
   252→      })
   253→      const created = await createRes.json()
   254→      if (!createRes.ok) throw new Error(created.error || 'Failed to create submission')
   255→      const submissionId = created.submission.id
   256→      // 2. Submit it (DRAFT → SUBMITTED)
   257→      const submitRes = await fetch(`/api/submissions/${submissionId}/submit`, {
   258→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   259→        body: JSON.stringify({ comment: `Submitted via Data Entry — ${activeSub.label}` }),
   260→      })
   261→      const submitted = await submitRes.json()
   262→      if (!submitRes.ok) throw new Error(submitted.error || 'Failed to submit submission')
   263→      setSubmitResult({ submissionId, status: submitted.submission?.status || 'SUBMITTED' })
   264→      setActionInfo('Submission sent for review. Status: SUBMITTED. You will be notified when BU review completes.')
   265→      refreshExistingRecords()
   266→    } catch (e: any) {
   267→      setActionError(e?.message || 'Submission failed')
   268→    } finally {
   269→      setSubmitting(false)
   270→    }
   271→  }
   272→
   273→  return (
   274→    <div className="space-y-5">
   275→      {/* HEADER */}
   276→      <motion.div
   277→        initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }}
   278→        transition={{ duration: 0.45, ease: [0.22, 1, 0.36, 1] }}
   279→        className="flex flex-col gap-3 md:flex-row md:items-end md:justify-between"
   280→      >
   281→        <div>
   282→          <div className="flex items-center gap-2">
   283→            <div className="kpi-tile bg-blue-50 text-blue-600"><FileText className="h-5 w-5" /></div>
   284→            <h1 className="text-2xl font-bold tracking-tight text-slate-800">Data Entry</h1>
   285→            <span className="status-pill status-submitted"><Clock className="h-3 w-3" /> Open</span>
   286→          </div>
   287→          <p className="mt-1 text-sm text-slate-500">
   288→            Source-record capture · {selectedProject?.projectCode} · {selectedPeriod?.label}
   289→            {readOnly && <span className="ml-2 inline-flex items-center gap-1 text-amber-700"><Lock className="h-3 w-3" /> Read-only — your role does not permit data entry</span>}
   290→          </p>
   291→        </div>
   292→
   293→        {/* Project + Period selectors */}
   294→        <div className="flex flex-wrap items-center gap-2">
   295→          <div className="glass-subtle flex items-center gap-1.5 rounded-full px-3 py-2 text-xs">
   296→            <Building2 className="h-3.5 w-3.5 text-blue-600" />
   297→            <select
   298→              value={selectedProjectId ?? ''} disabled={readOnly}
   299→              onChange={(e) => setSelectedProjectId(e.target.value)}
   300→              className="max-w-[160px] bg-transparent font-semibold text-slate-700 outline-none"
   301→            >
   302→              {projects.map(p => <option key={p.id} value={p.id}>{p.projectCode}</option>)}
   303→            </select>
   304→          </div>
   305→          <div className="glass-subtle flex items-center gap-1.5 rounded-full px-3 py-2 text-xs">
   306→            <CalendarClock className="h-3.5 w-3.5 text-blue-600" />
   307→            <select
   308→              value={selectedPeriodId ?? ''} disabled={readOnly}
   309→              onChange={(e) => setSelectedPeriodId(e.target.value)}
   310→              className="bg-transparent font-semibold text-slate-700 outline-none"
   311→            >
   312→              {periods.map(p => <option key={p.id} value={p.id}>{p.label}</option>)}
   313→            </select>
   314→          </div>
   315→        </div>
   316→      </motion.div>
   317→
   318→      {/* SUB-MODULE TAB BAR */}
   319→      <motion.div
   320→        initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }}
   321→        transition={{ duration: 0.4, delay: 0.05 }}
   322→        className="glass-nav flex items-center gap-0.5 overflow-x-auto scroll-elegant rounded-2xl p-1.5"
   323→      >
   324→        {SUB_MODULES.map(m => {
   325→          const active = m.key === subModule
   326→          const Icon = m.icon
   327→          return (
   328→            <button key={m.key} onClick={() => setDataEntrySubModule(m.key)}
   329→              className={`group relative flex items-center gap-1.5 whitespace-nowrap rounded-xl px-3 py-2 text-xs font-semibold transition ${active ? 'bg-white text-blue-700 shadow-sm' : 'text-slate-500 hover:bg-white/60 hover:text-slate-800'}`}>
   330→              <Icon className="h-3.5 w-3.5" /> {m.label}
   331→              {active && <motion.div layoutId="data-entry-underline" className="absolute -bottom-0.5 left-3 right-3 h-0.5 rounded-full bg-blue-500" />}
   332→            </button>
   333→          )
   334→        })}
   335→      </motion.div>
   336→
   337→      {/* STEPPER */}
   338→      <Stepper step={step} />
   339→
   340→      {/* MAIN GRID */}
   341→      <div className="grid gap-4 lg:grid-cols-3">
   342→        {/* FORM PANEL */}
   343→        <motion.section
   344→          initial={{ opacity: 0, y: 16 }} animate={{ opacity: 1, y: 0 }}
   345→          transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1], delay: 0.1 }}
   346→          className="glass glass-shimmer rounded-2xl p-4 lg:col-span-2"
   347→        >
   348→          <div className="mb-3 flex items-start justify-between">
   349→            <div className="flex items-start gap-2.5">
   350→              <div className="kpi-tile bg-blue-50 text-blue-600" style={{ width: 32, height: 32 }}>
   351→                <activeSub.icon className="h-4 w-4" />
   352→              </div>
   353→              <div>
   354→                <h3 className="text-sm font-bold text-slate-800">{activeSub.label} Entry Form</h3>
   355→                <p className="text-[11px] text-slate-500">
   356→                  {readOnly ? 'Read-only view — your role does not permit data entry' : 'Fill values, save draft, and the engine will validate + compute automatically.'}
   357→                </p>
   358→              </div>
   359→            </div>
   360→            <span className={`status-pill ${saved ? 'status-approved' : 'status-draft'}`}>
   361→              {saved ? 'Saved' : 'Draft'}
   362→            </span>
   363→          </div>
   364→
   365→          {/* Form body */}
   366→          <div className="space-y-3">
   367→            {readOnly && (
   368→              <div className="rounded-xl border border-amber-200/70 bg-amber-50/60 px-3 py-2 text-[11px] font-medium text-amber-800">
   369→                <Lock className="mr-1 inline h-3 w-3" /> Read-only — your role ({roleKey}) does not permit data entry. Switch to a writer role to edit.
   370→              </div>
   371→            )}
   372→
   373→            <AnimatePresence mode="wait">
   374→              <motion.div key={subModule}
   375→                initial={{ opacity: 0, x: 8 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -8 }}
   376→                transition={{ duration: 0.22 }}
   377→              >
   378→                {subModule === 'energy' && (
   379→                  <EnergyForm
   380→                    projectId={selectedProjectId!} periodId={selectedPeriodId!} evidence={evidence}
   381→                    readOnly={readOnly} formStateRef={formStateRef} onSaved={handleSaved}
   382→                  />
   383→                )}
   384→                {subModule === 'water' && (
   385→                  <WaterForm
   386→                    projectId={selectedProjectId!} periodId={selectedPeriodId!} evidence={evidence}
   387→                    readOnly={readOnly} formStateRef={formStateRef} onSaved={handleSaved}
   388→                  />
   389→                )}
   390→                {subModule === 'waste' && (
   391→                  <WasteForm
   392→                    projectId={selectedProjectId!} periodId={selectedPeriodId!} evidence={evidence}
   393→                    readOnly={readOnly} formStateRef={formStateRef} onSaved={handleSaved}
   394→                  />
   395→                )}
   396→                {subModule === 'workforce' && (
   397→                  <WorkforceForm
   398→                    projectId={selectedProjectId!} periodId={selectedPeriodId!} evidence={evidence}
   399→                    readOnly={readOnly} formStateRef={formStateRef} onSaved={handleSaved}
   400→                  />
   401→                )}
   402→                {subModule === 'safety' && (
   403→                  <SafetyForm
   404→                    projectId={selectedProjectId!} periodId={selectedPeriodId!} evidence={evidence}
   405→                    readOnly={readOnly} formStateRef={formStateRef} onSaved={handleSaved}
   406→                  />
   407→                )}
   408→                {subModule === 'travel' && (
   409→                  <div className="rounded-xl border border-dashed border-white/70 bg-white/40 px-4 py-10 text-center">
   410→                    <Plane className="mx-auto h-8 w-8 text-slate-400" />
   411→                    <div className="mt-2 text-sm font-bold text-slate-700">Travel &amp; Mobility Module</div>
   412→                    <div className="mt-1 text-xs text-slate-500">Business-trip emissions capture is coming soon. Use the Energy / Fuel module for vehicle fuel logs in the interim.</div>
   413→                  </div>
   414→                )}
   415→              </motion.div>
   416→            </AnimatePresence>
   417→          </div>
   418→
   419→          {/* Existing records */}
   420→          {existingRecords.length > 0 && (
   421→            <div className="mt-4">
   422→              <div className="mb-1.5 flex items-center justify-between">
   423→                <h4 className="text-[11px] font-bold uppercase tracking-wide text-slate-400">Existing records · {selectedPeriod?.label}</h4>
   424→                <span className="text-[10px] text-slate-400">{existingRecords.length} record(s)</span>
   425→              </div>
   426→              <div className="max-h-44 space-y-1 overflow-y-auto scroll-elegant pr-1">
   427→                {existingRecords.map((r: any) => (
   428→                  <div key={r.id} className="flex items-center justify-between rounded-lg bg-white/50 px-2.5 py-1.5 text-[11px]">
   429→                    <div className="flex min-w-0 items-center gap-2">
   430→                      <span className={`status-pill ${r.status === 'APPROVED' ? 'status-approved' : r.status === 'DRAFT' ? 'status-draft' : 'status-submitted'}`}>{r.status}</span>
   431→                      <span className="truncate font-semibold text-slate-700">{describeRecord(r, subModule)}</span>
   432→                    </div>
   433→                    <span className={`status-pill ${r.validationStatus === 'PASSED' ? 'status-approved' : r.validationStatus === 'FAILED' ? 'status-missing' : 'status-warning'}`}>{r.validationStatus}</span>
   434→                  </div>
   435→                ))}
   436→              </div>
   437→            </div>
   438→          )}
   439→          {existingRecords.length === 0 && !saved && !readOnly && (
   440→            <div className="mt-4 rounded-xl border border-dashed border-white/70 bg-white/40 px-4 py-6 text-center text-[11px] text-slate-500">

   647→function ValidationResults({ saved, validationRun }: { saved: SavedRecord | null; validationRun: { passed: number; errors: number; warnings: number } | null }) {
   648→  if (!saved) {
   649→    return (
   650→      <div className="mt-3 rounded-xl border border-dashed border-white/70 bg-white/40 px-3 py-4 text-center text-[11px] text-slate-400">

   661→    <div className="mt-3 space-y-2">
   662→      <div className="flex items-center gap-1.5">

   667→        ) : (
   668→          <span className="status-pill status-warning"><AlertTriangle className="h-3 w-3" /> {warnings.length} warning(s)</span>
   669→        )}
   670→        {saved.validationStatus && (
   671→          <span className="text-[10px] font-semibold uppercase tracking-wide text-slate-400">Status: {saved.validationStatus}</span>
   672→        )}
   673→      </div>
   674→
   675→      {validationRun && (
   676→        <div className="rounded-lg bg-blue-50/60 px-2.5 py-1.5 text-[10px] text-blue-700">
   677→          Validation run: {validationRun.passed} passed · {validationRun.warnings} warning(s) · {validationRun.errors} error(s)
   678→        </div>
   679→      )}
   680→
   681→      {issues.length === 0 ? (
   682→        <div className="rounded-lg border border-emerald-200/70 bg-emerald-50/50 px-3 py-2 text-[11px] text-emerald-700">
   683→          All validation rules passed. Record is eligible for submission.

   685→      ) : (
   686→        <div className="space-y-1.5">
   687→          {issues.map((iss, i) => <IssuePill key={i} issue={iss} delay={i * 0.04} />)}
   688→        </div>
   689→      )}
   690→    </div>
   691→  )

   763→function EnergyForm({ projectId, periodId, evidence, readOnly, formStateRef, onSaved }: FormProps) {
   764→  const key = `energy-${projectId}-${periodId}`
   765→  const initial = formStateRef.current[key] || {}
   766→  const [source, setSource] = useState<string>(initial.source ?? 'Grid Electricity')
   767→  const [quantity, setQuantity] = useState<string>(initial.quantity ?? '')
   768→  const [sourceUnit, setSourceUnit] = useState<string>(initial.sourceUnit ?? 'KWH')
   769→  const [vendor, setVendor] = useState<string>(initial.vendor ?? '')
   770→  const [meterRef, setMeterRef] = useState<string>(initial.meterRef ?? '')
   771→  const [evidenceId, setEvidenceId] = useState<string>(initial.evidenceId ?? '')
   772→  const [submitting, setSubmitting] = useState(false)
   773→  const [error, setError] = useState('')
   774→
   775→  const preset = ENERGY_SOURCES.find(s => s.source === source)
   776→  useEffect(() => {
   777→    if (preset) setSourceUnit(preset.unit)
   778→  }, [source, preset])
   779→
   780→  const persist = (patch: any) => { formStateRef.current[key] = { ...(formStateRef.current[key] || {}), ...patch } }
   781→  useEffect(() => { persist({ source, quantity, sourceUnit, vendor, meterRef, evidenceId }) }, [source, quantity, sourceUnit, vendor, meterRef, evidenceId])
   782→
   783→  const save = async () => {
   784→    setError('')
   785→    if (!quantity || Number(quantity) <= 0) { setError('Quantity must be > 0'); return }
   786→    setSubmitting(true)
   787→    try {
   788→      const r = await fetch('/api/energy', {
   789→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   790→        body: JSON.stringify({
   791→          projectId, reportingPeriodId: periodId,
   792→          source, sourceCategory: preset?.category ?? 'NON_RENEWABLE',
   793→          quantity: Number(quantity), sourceUnit, vendor: vendor || undefined,
   794→          meterRef: meterRef || undefined, evidenceId: evidenceId || undefined,
   795→        }),
   796→      })
   797→      const d = await r.json()
   798→      if (!r.ok) throw new Error(d.error || d.payload?.error || 'Save failed')
   799→      onSaved({
   800→        id: d.record.id, recordType: 'ENERGY',
   801→        issues: d.issues || [],
   802→        calculation: d.calculation || null,
   803→        validationStatus: d.record.validationStatus,
   804→      })
   805→    } catch (e: any) {
   806→      setError(e?.message || 'Save failed')
   807→    } finally {
   808→      setSubmitting(false)
   809→    }
   810→  }
   811→
   812→  return (
   813→    <FormGrid>
   814→      <FormField label="Energy source" hint="Auto-selects the category (renewable / non-renewable).">
   815→        <select value={source} disabled={readOnly} onChange={(e) => setSource(e.target.value)}
   816→          className={inputClass}>
   817→          {ENERGY_SOURCES.map(s => <option key={s.source} value={s.source}>{s.source}</option>)}
   818→        </select>
   819→      </FormField>
   820→      <FormField label="Source category" hint="Auto-derived from the chosen source.">
   821→        <input value={preset?.category ?? 'NON_RENEWABLE'} disabled className={inputClass + ' opacity-60'} />
   822→      </FormField>
   823→      <FormField label="Quantity" hint="Metered consumption for the period.">
   824→        <input type="number" min="0" step="any" value={quantity} disabled={readOnly}
   825→          onChange={(e) => setQuantity(e.target.value)} placeholder="e.g. 384000" className={inputClass} />
   826→      </FormField>
   827→      <FormField label="Source unit" hint="Standard UnitMaster codes only.">
   828→        <select value={sourceUnit} disabled={readOnly} onChange={(e) => setSourceUnit(e.target.value)} className={inputClass}>
   829→          {['KWH', 'MWH', 'GJ', 'L', 'KL', 'M3', 'KG', 'TON'].map(u => <option key={u} value={u}>{u}</option>)}
   830→        </select>
   831→      </FormField>
   832→      <FormField label="Vendor" hint="Utility / fuel supplier.">
   833→        <input value={vendor} disabled={readOnly} onChange={(e) => setVendor(e.target.value)} placeholder="e.g. TSSPDCL" className={inputClass} />
   834→      </FormField>
   835→      <FormField label="Meter reference" hint="Unique meter ID for this site.">
   836→        <input value={meterRef} disabled={readOnly} onChange={(e) => setMeterRef(e.target.value)} placeholder="e.g. MTR-GJT-33kV-01" className={inputClass} />
   837→      </FormField>
   838→      <FormField label="Evidence file" hint="Optional — pick from uploads." full>
   839→        <select value={evidenceId} disabled={readOnly} onChange={(e) => setEvidenceId(e.target.value)} className={inputClass}>
   840→          <option value="">— None —</option>
   841→          {evidence.map(ev => <option key={ev.id} value={ev.id}>{ev.fileName}</option>)}
   842→        </select>
   843→      </FormField>
   844→      <LiveEstimateCard source={source} quantity={quantity} sourceUnit={sourceUnit} />
   845→      <FormActions error={error} submitting={submitting} readOnly={readOnly} onSave={save} />
   846→    </FormGrid>
   847→  )
   848→}

   852→function LiveEstimateCard({ source, quantity, sourceUnit }: { source: string; quantity: string; sourceUnit: string }) {
   853→  const qty = Number(quantity)
   854→  const valid = qty > 0 && quantity !== ''
   855→  // Match the seeded emission factors (src/app/api/energy/route.ts findEmissionFactorForSource)
   856→  const FACTORS: Record<string, { value: number; unit: string; scope: string; methodology: string; gjPerUnit: number }> = {
   857→    'Grid Electricity': { value: 0.716, unit: 'kgCO2e/kWh', scope: 'SCOPE_2', methodology: 'CEA v19', gjPerUnit: sourceUnit === 'MWH' ? 3.6 : 0.0036 },
   858→    'Diesel (HSD)': { value: 2.637, unit: 'kgCO2e/L', scope: 'SCOPE_1', methodology: 'IPCC 2006', gjPerUnit: sourceUnit === 'KL' ? 38.3 : 0.0383 },
   859→    'Petrol (MS)': { value: 2.296, unit: 'kgCO2e/L', scope: 'SCOPE_1', methodology: 'IPCC 2006', gjPerUnit: 0.0348 },
   860→    'Coal (Sub-bituminous)': { value: 1.9, unit: 'kgCO2e/kg', scope: 'SCOPE_1', methodology: 'IPCC 2006', gjPerUnit: 0.0227 },
   861→    'CNG': { value: 2.19, unit: 'kgCO2e/kg', scope: 'SCOPE_1', methodology: 'IPCC 2006', gjPerUnit: 0.05 },

   864→  }
   865→  const f = FACTORS[source]
   866→  if (!f || !valid) {
   867→    return (

   871→    )
   872→  }
   873→  const co2eKg = qty * f.value
   874→  const co2eT = co2eKg / 1000
   875→  const energyGJ = qty * f.gjPerUnit
   876→  const scopeColor = f.scope === 'SCOPE_1' ? 'bg-rose-50 text-rose-600' : 'bg-blue-50 text-blue-600'
   877→  return (
   878→    <motion.div initial={{ opacity: 0, y: 6 }} animate={{ opacity: 1, y: 0 }} className="md:col-span-2 rounded-xl border border-blue-200/70 bg-gradient-to-br from-blue-50/70 to-cyan-50/40 px-4 py-3">

   881→          <span className="flex h-6 w-6 items-center justify-center rounded-md bg-blue-500 text-white"><Calculator className="h-3.5 w-3.5" /></span>
   882→          <span className="text-[11px] font-bold uppercase tracking-wide text-slate-600">Live emission estimate</span>
   883→          <span className={`status-pill ${scopeColor} !text-[9px]`}>{f.scope}</span>
   884→        </div>
   885→        <span className="text-[10px] text-slate-400">factor v1 · {f.methodology}</span>
   886→      </div>
   887→      <div className="mt-2 grid grid-cols-3 gap-3">
   888→        <div>
   889→          <div className="text-[9px] font-semibold uppercase tracking-wide text-slate-400">Estimated CO₂e</div>

  1270→function SafetyForm({ projectId, periodId, evidence, readOnly, formStateRef, onSaved }: FormProps) {
  1271→  const key = `safety-${projectId}-${periodId}`
  1272→  const initial = formStateRef.current[key] || {}
  1273→  const [recordType, setRecordType] = useState<string>(initial.recordType ?? 'INCIDENT')
  1274→  const [fatalities, setFatalities] = useState<string>(initial.fatalities ?? '0')
  1275→  const [injuries, setInjuries] = useState<string>(initial.injuries ?? '0')
  1276→  const [lostTimeIncidents, setLostTimeIncidents] = useState<string>(initial.lostTimeIncidents ?? '0')
  1277→  const [recordableInjuries, setRecordableInjuries] = useState<string>(initial.recordableInjuries ?? '0')
  1278→  const [highConsequenceIncidents, setHighConsequenceIncidents] = useState<string>(initial.highConsequenceIncidents ?? '0')
  1279→  const [trainingHours, setTrainingHours] = useState<string>(initial.trainingHours ?? '')
  1280→  const [safetyHours, setSafetyHours] = useState<string>(initial.safetyHours ?? '')
  1281→  const [manHoursWorked, setManHoursWorked] = useState<string>(initial.manHoursWorked ?? '')
  1282→  const [correctiveActions, setCorrectiveActions] = useState<string>(initial.correctiveActions ?? '')
  1283→  const [evidenceId, setEvidenceId] = useState<string>(initial.evidenceId ?? '')
  1284→  const [submitting, setSubmitting] = useState(false)
  1285→  const [error, setError] = useState('')
  1286→
  1287→  const lti = Number(lostTimeIncidents) || 0
  1288→  const mhw = Number(manHoursWorked) || 0
  1289→  const ltifr = mhw > 0 ? ((lti * 1000000) / mhw).toFixed(3) : null
  1290→
  1291→  const persist = (patch: any) => { formStateRef.current[key] = { ...(formStateRef.current[key] || {}), ...patch } }
  1292→  useEffect(() => { persist({ recordType, fatalities, injuries, lostTimeIncidents, recordableInjuries, highConsequenceIncidents, trainingHours, safetyHours, manHoursWorked, correctiveActions, evidenceId }) }, [recordType, fatalities, injuries, lostTimeIncidents, recordableInjuries, highConsequenceIncidents, trainingHours, safetyHours, manHoursWorked, correctiveActions, evidenceId])
  1293→

  1299→      const r = await fetch('/api/safety', {
  1300→        method: 'POST', headers: { 'Content-Type': 'application/json' },
  1301→        body: JSON.stringify({
  1302→          projectId, reportingPeriodId: periodId, recordType,
  1303→          fatalities: Number(fatalities) || 0, injuries: Number(injuries) || 0,
  1304→          lostTimeIncidents: Number(lostTimeIncidents) || 0,
  1305→          recordableInjuries: Number(recordableInjuries) || 0,
  1306→          highConsequenceIncidents: Number(highConsequenceIncidents) || 0,
  1307→          trainingHours: trainingHours ? Number(trainingHours) : undefined,
  1308→          safetyHours: safetyHours ? Number(safetyHours) : undefined,
  1309→          manHoursWorked: manHoursWorked ? Number(manHoursWorked) : undefined,
  1310→          correctiveActions: correctiveActions || undefined,
  1311→          evidenceId: evidenceId || undefined,
  1312→        }),
  1313→      })
  1314→      const d = await r.json()
  1315→      if (!r.ok) throw new Error(d.error || d.payload?.error || 'Save failed')
  1316→      onSaved({
  1317→        id: d.record.id, recordType: 'SAFETY',
  1318→        issues: d.issues || [], calculation: null,
  1319→        derivedLtifr: d.derivedLtifr ?? null,
  1320→        validationStatus: d.record.validationStatus,
  1321→      })
  1322→    } catch (e: any) {
  1323→      setError(e?.message || 'Save failed')

  1440→function describeRecord(r: any, subModule: SubModule): string {
  1441→  if (subModule === 'energy') return `${r.source} · ${r.quantity} ${r.sourceUnit}`
  1442→  if (subModule === 'water') return `${r.source} · ${r.withdrawal} ${r.sourceUnit}`
  1443→  if (subModule === 'waste') return `${r.wasteType} · ${r.generatedQty} ${r.sourceUnit}`
  1444→  if (subModule === 'workforce') return `${r.category} · ${(r.permanent || 0) + (r.nonPermanent || 0)} people`
  1445→  if (subModule === 'safety') return `${r.recordType} · ${r.manHoursWorked ?? 0} hrs`
  1446→  return r.id?.slice(-8) ?? 'Record'
  1447→}

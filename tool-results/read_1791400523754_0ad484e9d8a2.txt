   138→   138→function timeAgo(iso: string): string {
   139→   139→  const d = new Date(iso)
   140→   140→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   141→   141→  if (s < 30) return 'just now'
   142→   142→  if (s < 60) return `${s}s ago`
   143→   143→  const m = Math.floor(s / 60)
   144→   144→  if (m < 60) return `${m}m ago`
   145→   145→  const h = Math.floor(m / 60)
   146→   146→  if (h < 24) return `${h}h ago`
   147→   147→  const dd = Math.floor(h / 24)
   148→   148→  return `${dd}d ago`
   149→   149→}

   209→   209→/** Build a monthly incident trend from totals + periods. */
   210→   210→function buildIncidentTrend(
   211→   211→  periods: { id: string; label: string }[] | undefined,
   212→   212→  k: Pick<Kpis, 'injuries' | 'lti' | 'fatalities'>,
   213→   213→): { label: string; injuries: number; lti: number; fatalities: number }[] {
   214→   214→  const totalInjuries = Math.max(k.injuries, 0)
   215→   215→  const totalLti = Math.max(k.lti, 0)
   216→   216→  const totalFatal = Math.max(k.fatalities, 0)
   217→   217→
   218→   218→  let labels: string[]
   219→   219→  if (periods && periods.length > 0) {
   220→   220→    labels = periods.slice(-6).map(p => p.label)
   221→   221→  } else {
   222→   222→    const now = new Date()
   223→   223→    labels = Array.from({ length: 6 }).map((_, i) => {
   224→   224→      const d = new Date(now.getFullYear(), now.getMonth() - (5 - i), 1)
   225→   225→      return d.toLocaleString(undefined, { month: 'short' })
   226→   226→    })
   227→   227→  }
   228→   228→
   229→   229→  // Distribute totals deterministically — latest period holds the bulk.
   230→   230→  const weights = [0.05, 0.08, 0.12, 0.15, 0.20, 0.40]
   231→   231→  return labels.map((label, i) => {
   232→   232→    const w = weights[i] ?? weights[weights.length - 1]
   233→   233→    return {
   234→   234→      label,
   235→   235→      injuries: Math.round(totalInjuries * w),
   236→   236→      lti: Math.round(totalLti * w),
   237→   237→      fatalities: i === labels.length - 1 ? totalFatal : 0,
   238→   238→    }
   239→   239→  })
   240→   240→}

   242→   242→/** Derive a deterministic incident roster from KPI totals. */
   243→   243→function deriveIncidentRoster(k: Kpis): {
   244→   244→  id: string
   245→   245→  date: string
   246→   246→  type: 'Injury' | 'LTI' | 'Near Miss' | 'Fatality' | 'Property Damage'
   247→   247→  severity: 'Critical' | 'High' | 'Medium' | 'Low'
   248→   248→  location: string
   249→   249→  status: 'Open' | 'Under Review' | 'Resolved' | 'Closed'
   250→   250→  owner: string
   251→   251→}[] {
   252→   252→  const total = Math.min(12, Math.max(6, k.injuries + k.lti + k.fatalities + 3))
   253→   253→  const locations = ['Plant A — Mumbai', 'Plant B — Pune', 'Site Office — Nagpur',
   254→   254→    'Warehouse — Nashik', 'Project Site — Thane', 'Logistics Hub — Vashi']
   255→   255→  const owners = ['R. Sharma (Safety Officer)', 'A. Iyer (Site In-charge)',
   256→   256→    'P. Nair (EHS Lead)', 'S. Reddy (Maintenance)', 'M. Patel (Operations)',
   257→   257→    'K. Rao (HSE Manager)']
   258→   258→  const types: Array<'Injury' | 'LTI' | 'Near Miss' | 'Fatality' | 'Property Damage'> =
   259→   259→    ['Injury', 'Near Miss', 'LTI', 'Property Damage', 'Injury', 'Near Miss']
   260→   260→  const severities: Array<'Critical' | 'High' | 'Medium' | 'Low'> =
   261→   261→    ['High', 'Medium', 'Low', 'Critical', 'Medium', 'Low']
   262→   262→  const statuses: Array<'Open' | 'Under Review' | 'Resolved' | 'Closed'> =
   263→   263→    ['Open', 'Under Review', 'Resolved', 'Closed', 'Open', 'Under Review']
   264→   264→
   265→   265→  const seed = (k.injuries + k.lti + k.fatalities) || 1
   266→   266→  const rows: ReturnType<typeof deriveIncidentRoster> = []
   267→   267→  for (let i = 0; i < total; i++) {
   268→   268→    const r = ((seed * (i + 7)) % 997) / 997
   269→   269→    const idx = (i + Math.floor(r * 3)) % 6
   270→   270→    const dayOffset = (i + 1) * 86400_000 * (i % 5 + 1)
   271→   271→    rows.push({
   272→   272→      id: `INC-${(2400 + i).toString()}`,
   273→   273→      date: new Date(Date.now() - dayOffset).toISOString(),
   274→   274→      type: types[idx],
   275→   275→      severity: severities[idx],
   276→   276→      location: locations[i % locations.length],
   277→   277→      status: statuses[idx],
   278→   278→      owner: owners[i % owners.length],
   279→   279→    })
   280→   280→  }
   281→   281→  return rows
   282→   282→}

   315→   315→      findings,
   316→   316→      critical,
   317→   317→      score,
   318→   318→      status,
   319→   319→      date: new Date(Date.now() - i * 86400_000 * 3).toISOString(),
   320→   320→    })
   321→   321→  }
   322→   322→  return inspections
   323→   323→}

   368→   368→/** Derive environmental records from KPIs + periods. */
   369→   369→function deriveEnvironmentalRecords(
   370→   370→  k: Kpis,
   371→   371→  periods: { id: string; label: string }[] | undefined,
   372→   372→): { period: string; water: number; waste: number; recycled: number; hazardous: number }[] {
   373→   373→  const labels = periods && periods.length > 0
   374→   374→    ? periods.slice(-6).map(p => p.label)
   375→   375→    : ['Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep']
   376→   376→  const seed = (k.waterWithdrawalKL + k.wasteGeneratedT) || 1
   377→   377→  return labels.map((label, i) => {
   378→   378→    const r = ((seed * (i + 17)) % 997) / 997
   379→   379→    const w = Math.max(1, Math.round(k.waterWithdrawalKL * (0.10 + r * 0.18)))
   380→   380→    const waste = Math.max(0.5, Math.round(k.wasteGeneratedT * (0.10 + r * 0.20)))
   381→   381→    const recycled = Math.round(waste * (k.wasteRecycledShare / 100))
   382→   382→    const hazardous = Math.round(k.hazardousWasteT * (0.10 + r * 0.25))
   383→   383→    return { period: label, water: w, waste, recycled, hazardous }
   384→   384→  })
   385→   385→}

  1815→  1815→export function EhsWorkspace() {
  1816→  1816→  const { activeModule } = useApp()

  1818→  1818→  const [activities, setActivities] = useState<ActivityItem[]>([])
  1819→  1819→  const [tasks, setTasks] = useState<ActionItem[]>([])
  1820→  1820→  const [loading, setLoading] = useState(true)
  1821→  1821→  const [activityLoading, setActivityLoading] = useState(true)
  1822→  1822→  const [tasksLoading, setTasksLoading] = useState(true)
  1823→  1823→  const [error, setError] = useState('')
  1824→  1824→  const mountedRef = useRef(true)
  1825→  1825→
  1826→  1826→  /* ---- fetchers ---- */
  1827→  1827→  const fetchOverview = useCallback(async () => {
  1828→  1828→    try {
  1829→  1829→      const res = await fetch('/api/overview', { cache: 'no-store' })
  1830→  1830→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1831→  1831→      const data = (await res.json()) as OverviewData
  1832→  1832→      if (!mountedRef.current) return
  1833→  1833→      setOverview(data)
  1834→  1834→      setError('')
  1835→  1835→    } catch (e) {
  1836→  1836→      if (!mountedRef.current) return
  1837→  1837→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
  1838→  1838→    }
  1839→  1839→  }, [overview])
  1840→  1840→
  1841→  1841→  const fetchActivities = useCallback(async () => {
  1842→  1842→    try {
  1843→  1843→      setActivityLoading(true)
  1844→  1844→      const res = await fetch('/api/activity?take=10', { cache: 'no-store' })
  1845→  1845→      if (!res.ok) throw new Error(`HTTP ${res.status}`)

  1847→  1847→      if (!mountedRef.current) return

  1849→  1849→      setActivities(filtered)
  1850→  1850→    } catch {
  1851→  1851→      /* silent — keep existing feed on poll error */
  1852→  1852→    } finally {
  1853→  1853→      if (mountedRef.current) setActivityLoading(false)
  1854→  1854→    }
  1855→  1855→  }, [])
  1856→  1856→
  1857→  1857→  const fetchTasks = useCallback(async () => {
  1858→  1858→    try {
  1859→  1859→      setTasksLoading(true)
  1860→  1860→      const res = await fetch('/api/action-items', { cache: 'no-store' })
  1861→  1861→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1862→  1862→      const data = (await res.json()) as ActionItemsResponse
  1863→  1863→      if (!mountedRef.current) return
  1864→  1864→      const filtered = (Array.isArray(data.tasks) ? data.tasks : []).filter(isEhsTask).slice(0, 8)
  1865→  1865→      setTasks(filtered)
  1866→  1866→    } catch {
  1867→  1867→      /* silent */
  1868→  1868→    } finally {
  1869→  1869→      if (mountedRef.current) setTasksLoading(false)
  1870→  1870→    }
  1871→  1871→  }, [])
  1872→  1872→
  1873→  1873→  /* ---- initial load ---- */

  1876→  1876→    ;(async () => {

  1879→  1879→      if (mountedRef.current) setLoading(false)
  1880→  1880→    })()

  1882→  1882→  }, [fetchOverview, fetchActivities, fetchTasks])
  1883→  1883→

  1886→  1886→    const activityTimer = setInterval(fetchActivities, 30_000)

  1891→  1891→      clearInterval(overviewTimer)
  1892→  1892→      clearInterval(tasksTimer)
  1893→  1893→    }
  1894→  1894→  }, [fetchActivities, fetchOverview, fetchTasks])
  1895→  1895→

  1898→  1898→    return overview.kpis

  1900→  1900→

  1904→  1904→  }, [overview])

  1911→  1911→  }

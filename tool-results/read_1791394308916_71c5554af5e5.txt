   120→function timeAgo(iso: string): string {
   121→  const d = new Date(iso)
   122→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   123→  if (s < 30) return 'just now'
   124→  if (s < 60) return `${s}s ago`
   125→  const m = Math.floor(s / 60)
   126→  if (m < 60) return `${m}m ago`
   127→  const h = Math.floor(m / 60)
   128→  if (h < 24) return `${h}h ago`
   129→  const dd = Math.floor(h / 24)
   130→  return `${dd}d ago`
   131→}

   133→function initials(name: string): string {
   134→  return name.split(/\s+/).filter(Boolean).slice(0, 2).map(s => s[0]?.toUpperCase() ?? '').join('') || '?'
   135→}

   137→function formatNumber(n: number, digits = 1): string {
   138→  if (!isFinite(n)) return '0'
   139→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   140→  return n.toFixed(digits)
   141→}

   143→function statusClass(status?: string | null): string {
   144→  switch ((status ?? '').toUpperCase()) {
   145→    case 'APPROVED': case 'COMPLETED': return 'status-approved'
   146→    case 'SUBMITTED': return 'status-submitted'
   147→    case 'UNDER_REVIEW': case 'REVIEW': return 'status-review'
   148→    case 'DRAFT': return 'status-draft'
   149→    case 'LOCKED': return 'status-locked'
   150→    case 'MISSING': case 'ERROR': case 'BLOCKING': return 'status-missing'
   151→    case 'WARNING': return 'status-warning'
   152→    case 'EVIDENCE_VERIFIED': case 'VERIFIED': return 'status-verified'
   153→    default: return 'status-draft'
   154→  }
   155→}

   157→/** Filter activities to HR / people / workforce actions. */
   158→function isHrActivity(a: ActivityItem): boolean {
   159→  const mod = (a.module ?? '').toUpperCase()
   160→  const act = (a.action ?? '').toUpperCase()
   161→  const title = (a.title ?? '').toLowerCase()
   162→  const desc = (a.description ?? '').toLowerCase()
   163→  const hrModules = ['PEOPLE', 'WORKFORCE', 'HR']
   164→  const hrActions = ['WORKFORCE', 'TRAINING', 'EMPLOYEE', 'HIRING', 'ONBOARDING', 'DIVERSITY']
   165→  const hrKeywords = ['employee', 'workforce', 'training', 'worker', 'gender', 'diversity', 'hired', 'onboard', 'hr ']
   166→  return (
   167→    hrModules.includes(mod) ||
   168→    hrActions.some(k => act.includes(k)) ||
   169→    hrKeywords.some(k => title.includes(k) || desc.includes(k))
   170→  )
   171→}
   172→
   173→/** Filter action-items relevant to the HR user role. */
   174→function isHrTask(t: ActionItem): boolean {
   175→  const m = (t.module ?? '').toLowerCase()
   176→  const type = (t.type ?? '').toUpperCase()
   177→  const desc = (t.description ?? '').toLowerCase()
   178→  const title = (t.title ?? '').toLowerCase()
   179→  return (
   180→    m === 'submissions' ||
   181→    type === 'CORRECTION' ||
   182→    type === 'DRAFT_SUBMISSION' ||
   183→    type === 'BRSR_GAP' ||
   184→    title.includes('workforce') || title.includes('employee') ||
   185→    desc.includes('workforce') || desc.includes('employee') || desc.includes('training')
   186→  )
   187→}

   716→  const [activities, setActivities] = useState<ActivityItem[]>([])
   717→  const [tasks, setTasks] = useState<ActionItem[]>([])
   718→  const [loading, setLoading] = useState(true)
   719→  const [error, setError] = useState('')
   720→  const mountedRef = useRef(true)
   721→
   722→  /* ---- fetchers ---- */
   723→  const fetchOverview = useCallback(async () => {
   724→    try {
   725→      const res = await fetch('/api/overview', { cache: 'no-store' })
   726→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   727→      const data = (await res.json()) as OverviewData
   728→      if (!mountedRef.current) return
   729→      setOverview(data)
   730→      setError('')
   731→    } catch (e) {
   732→      if (!mountedRef.current) return
   733→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
   734→    }
   735→  }, [overview])
   736→
   737→  const fetchActivities = useCallback(async () => {
   738→    try {
   739→      const res = await fetch('/api/activity?take=10', { cache: 'no-store' })
   740→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   741→      const data = (await res.json()) as ActivityResponse
   742→      if (!mountedRef.current) return
   743→      const filtered = (Array.isArray(data.items) ? data.items : []).filter(isHrActivity).slice(0, 5)
   744→      setActivities(filtered)
   745→    } catch {
   746→      /* silent — keep existing feed on poll error */
   747→    }
   748→  }, [])
   749→
   750→  const fetchTasks = useCallback(async () => {
   751→    try {
   752→      const res = await fetch('/api/action-items', { cache: 'no-store' })
   753→      if (!res.ok) throw new Error(`HTTP ${res.status}`)

   755→      if (!mountedRef.current) return
   756→      const filtered = (Array.isArray(data.tasks) ? data.tasks : []).filter(isHrTask).slice(0, 6)
   757→      setTasks(filtered)
   758→    } catch {
   759→      /* silent */
   760→    }
   761→  }, [])
   762→
   763→  /* ---- initial load ---- */
   764→  useEffect(() => {
   765→    mountedRef.current = true
   766→    ;(async () => {
   767→      setLoading(true)
   768→      await Promise.all([fetchOverview(), fetchActivities(), fetchTasks()])
   769→      if (mountedRef.current) setLoading(false)
   770→    })()
   771→    return () => { mountedRef.current = false }
   772→  }, [])
   773→
   774→  /* ---- polling: activity every 30s, overview every 60s, tasks every 60s ---- */
   775→  useEffect(() => {
   776→    const activityTimer = setInterval(fetchActivities, 30_000)
   777→    const overviewTimer = setInterval(fetchOverview, 60_000)
   778→    const tasksTimer = setInterval(fetchTasks, 60_000)
   779→    return () => {
   780→      clearInterval(activityTimer)
   781→      clearInterval(overviewTimer)
   782→      clearInterval(tasksTimer)
   783→    }
   784→  }, [fetchActivities, fetchOverview, fetchTasks])
   785→
   786→  const k = useMemo<Kpis | null>(() => {
   787→    if (!overview?.kpis) return null
   788→    return overview.kpis
   789→  }, [overview])
   790→
   791→  const completionPct = k?.completion ?? 0
   792→  const totalWorkforce = k?.totalWorkforce ?? 0
   793→
   794→  // ---- Loading skeleton ----
   795→  if (loading && !overview) {
   796→    return (
   797→      <div className="space-y-5">
   798→        <div className="glass rounded-[20px] h-16 animate-pulse" />
   799→        <div className="grid grid-cols-1 gap-5 lg:grid-cols-[1fr_minmax(320px,35%)]">
   800→          <div className="space-y-5">
   801→            <div className="grid grid-cols-2 md:grid-cols-4 gap-3">

   803→                <div key={i} className="glass rounded-2xl h-[100px] animate-pulse" />
   804→              ))}
   805→            </div>
   806→            <div className="glass rounded-[20px] h-[320px] animate-pulse" />
   807→            <div className="glass rounded-[20px] h-[280px] animate-pulse" />
   808→          </div>
   809→          <div className="space-y-5">
   810→            <div className="glass rounded-[20px] h-[320px] animate-pulse" />
   811→            <div className="glass rounded-[20px] h-[280px] animate-pulse" />
   812→            <div className="glass rounded-[20px] h-[280px] animate-pulse" />
   813→          </div>
   814→        </div>
   815→      </div>
   816→    )
   817→  }
   818→
   819→  // ---- Error state ----
   820→  if (error && !overview) {
   821→    return (
   822→      <div className="glass rounded-[20px] p-10 flex flex-col items-center justify-center text-center min-h-[400px]">

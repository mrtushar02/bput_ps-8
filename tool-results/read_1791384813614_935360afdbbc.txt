   172→function timeAgo(iso: string): string {
   173→  const d = new Date(iso)
   174→  const s = Math.floor((Date.now() - d.getTime()) / 1000)
   175→  if (s < 30) return 'just now'
   176→  if (s < 60) return `${s}s ago`
   177→  const m = Math.floor(s / 60)
   178→  if (m < 60) return `${m}m ago`
   179→  const h = Math.floor(m / 60)
   180→  if (h < 24) return `${h}h ago`
   181→  const dd = Math.floor(h / 24)
   182→  return `${dd}d ago`
   183→}

   189→function statusClass(status?: string | null): string {
   190→  switch ((status ?? '').toUpperCase()) {
   191→    case 'APPROVED': return 'status-approved'
   192→    case 'SUBMITTED': return 'status-submitted'
   193→    case 'UNDER_REVIEW': case 'REVIEW': return 'status-review'
   194→    case 'DRAFT': return 'status-draft'
   195→    case 'LOCKED': return 'status-locked'
   196→    case 'MISSING': return 'status-missing'
   197→    case 'ERROR': case 'BLOCKING': return 'status-error'
   198→    case 'WARNING': return 'status-warning'
   199→    case 'EVIDENCE_VERIFIED': case 'VERIFIED': return 'status-verified'
   200→    case 'COMPLETED': return 'status-approved'
   201→    default: return 'status-draft'
   202→  }

   205→function formatNumber(n: number, digits = 1): string {
   206→  if (!isFinite(n)) return '0'
   207→  if (n >= 1000) return (n / 1000).toFixed(digits) + 'k'
   208→  return n.toFixed(digits)

   211→/** Derive per-module completion % from submissions, falling back to KPI soft values. */
   212→function moduleCompletion(subs: SubmissionItem[], kpis?: Kpis): { label: string; pct: number; tone: string }[] {
   213→  const groups: Record<string, { total: number; sum: number }> = {}
   214→  for (const s of subs) {
   215→    const k = (s.module || 'other').toLowerCase()
   216→    if (!groups[k]) groups[k] = { total: 0, sum: 0 }
   217→    groups[k].total += 1
   218→    groups[k].sum += s.completionPct || 0
   219→  }
   220→  const energy = groups['energy'] ? groups['energy'].sum / groups['energy'].total : (kpis?.renewableShare ?? 0)
   221→  const water = groups['water'] ? groups['water'].sum / groups['water'].total : (kpis?.waterRecycledShare ?? 0)
   222→  const waste = groups['waste'] ? groups['waste'].sum / groups['waste'].total : (kpis?.wasteRecycledShare ?? 0)
   223→  const safety = groups['safety'] ? groups['safety'].sum / groups['safety'].total : (kpis && kpis.ltifr >= 0 ? Math.max(0, 100 - kpis.ltifr * 5) : 80)
   224→  const workforce = groups['workforce'] ? groups['workforce'].sum / groups['workforce'].total : (kpis && kpis.trainingHours > 0 ? 88 : 70)
   225→  return [
   226→    { label: 'Energy', pct: Math.round(energy), tone: 'bg-blue-500' },
   227→    { label: 'Water', pct: Math.round(water), tone: 'bg-cyan-500' },
   228→    { label: 'Waste', pct: Math.round(waste), tone: 'bg-emerald-500' },
   229→    { label: 'Safety', pct: Math.round(safety), tone: 'bg-amber-500' },
   230→    { label: 'Workforce', pct: Math.round(workforce), tone: 'bg-violet-500' },
   231→  ]
   232→}

   323→  data, onLog,
   324→}: {
   325→  data: OverviewData
   326→  onLog: () => void
   327→}) {
   328→  const k = data.kpis
   329→  const trendEntries = Object.entries(data.trends || {})
   330→  const emissionsSeries = trendEntries.map(([label, t]) => ({ label, value: t.emissions }))
   331→  const waterSeries = trendEntries.map(([label, t]) => ({ label, value: t.water }))
   332→
   333→  // simple delta for trend pills — last vs prev
   334→  const delta = (series: { value: number }[]) => {
   335→    if (series.length < 2) return { dir: 'neutral' as const, text: '—' }
   336→    const last = series[series.length - 1].value
   337→    const prev = series[series.length - 2].value
   338→    if (prev === 0) return { dir: 'neutral' as const, text: '0%' }
   339→    const pct = ((last - prev) / Math.abs(prev)) * 100
   340→    return {
   341→      dir: pct >= 0 ? ('up' as const) : ('down' as const),
   342→      text: `${Math.abs(pct).toFixed(1)}%`,
   343→    }
   344→  }
   345→
   346→  return (
   347→    <motion.section
   348→      custom={0}
   349→      variants={cardEnter}

   973→export function SiteUserOverview() {
   974→  const { setActiveModule } = useApp()
   975→  const [overview, setOverview] = useState<OverviewData | null>(null)
   976→  const [activities, setActivities] = useState<ActivityItem[]>([])
   977→  const [submissions, setSubmissions] = useState<SubmissionItem[]>([])
   978→  const [loading, setLoading] = useState(true)
   979→  const [error, setError] = useState('')
   980→  const mountedRef = useRef(true)
   981→
   982→  /* ---- fetchers ---- */
   983→  const fetchOverview = useCallback(async () => {
   984→    try {
   985→      const res = await fetch('/api/overview', { cache: 'no-store' })
   986→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   987→      const data = (await res.json()) as OverviewData
   988→      if (!mountedRef.current) return
   989→      setOverview(data)
   990→      setError('')
   991→    } catch (e) {
   992→      if (!mountedRef.current) return
   993→      if (!overview) setError(e instanceof Error ? e.message : 'Failed to load overview')
   994→    }
   995→  }, [overview])
   996→
   997→  const fetchActivities = useCallback(async () => {
   998→    try {
   999→      const res = await fetch('/api/activity?take=20', { cache: 'no-store' })
  1000→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1001→      const data = (await res.json()) as ActivityResponse
  1002→      if (!mountedRef.current) return
  1003→      setActivities(Array.isArray(data.items) ? data.items : [])
  1004→    } catch {
  1005→      /* silent — keep existing feed on poll error */
  1006→    }
  1007→  }, [])
  1008→
  1009→  const fetchSubmissions = useCallback(async () => {
  1010→    try {
  1011→      const res = await fetch('/api/submissions?take=10', { cache: 'no-store' })
  1012→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1013→      const data = (await res.json()) as SubmissionResponse
  1014→      if (!mountedRef.current) return
  1015→      setSubmissions(Array.isArray(data.items) ? data.items : [])
  1016→    } catch {
  1017→      /* silent */
  1018→    }
  1019→  }, [])
  1020→
  1021→  /* ---- initial load ---- */
  1022→  useEffect(() => {
  1023→    mountedRef.current = true
  1024→    ;(async () => {
  1025→      setLoading(true)
  1026→      await Promise.all([fetchOverview(), fetchActivities(), fetchSubmissions()])
  1027→      if (mountedRef.current) setLoading(false)
  1028→    })()
  1029→    return () => { mountedRef.current = false }
  1030→  }, [])
  1031→
  1032→  /* ---- polling: activity every 30s, overview every 60s ---- */
  1033→  useEffect(() => {
  1034→    const activityTimer = setInterval(fetchActivities, 30_000)
  1035→    const overviewTimer = setInterval(fetchOverview, 60_000)
  1036→    return () => {
  1037→      clearInterval(activityTimer)
  1038→      clearInterval(overviewTimer)
  1039→    }
  1040→  }, [fetchActivities, fetchOverview])
  1041→
  1042→  const trendsArr = useMemo<Array<{ label: string } & Trend>>(() => {
  1043→    if (!overview?.trends) return []
  1044→    return Object.entries(overview.trends).map(([label, t]) => ({ label, ...t }))
  1045→  }, [overview])
  1046→
  1047→  // Guard: still loading initial data
  1048→  if (loading && !overview) {
  1049→    return (
  1050→      <div className="space-y-5">
  1051→        <div className="grid grid-cols-1 gap-5 lg:grid-cols-[1fr_minmax(380px,42%)]">
  1052→          <div className="space-y-5">
  1053→            <div className="glass rounded-[20px] h-[420px] animate-pulse" />
  1054→            <div className="glass rounded-[20px] h-[420px] animate-pulse" />
  1055→          </div>
  1056→          <div className="space-y-5">
  1057→            <div className="glass rounded-[20px] h-[420px] animate-pulse" />
  1058→            <div className="glass rounded-[20px] h-[420px] animate-pulse" />
  1059→          </div>
  1060→        </div>
  1061→        <div className="glass rounded-[20px] h-[280px] animate-pulse" />
  1062→        <div className="grid grid-cols-1 gap-5 lg:grid-cols-2">
  1063→          <div className="glass rounded-[20px] h-[260px] animate-pulse" />
  1064→          <div className="glass rounded-[20px] h-[260px] animate-pulse" />
  1065→        </div>
  1066→      </div>
  1067→    )
  1068→  }
  1069→
  1070→  // Guard: error and no data at all
  1071→  if (error && !overview) {
  1072→    return (
  1073→      <div className="glass rounded-[20px] p-10 flex flex-col items-center justify-center text-center min-h-[400px]">
  1074→        <AlertCircle className="h-10 w-10 text-rose-400 mb-3" />
  1075→        <p className="text-[14px] font-semibold text-slate-700 mb-1">Unable to load dashboard</p>
  1076→        <p className="text-[12px] text-slate-700 mb-4">{error}</p>
  1077→        <button

  1080→        >
  1081→          <RefreshCw className="h-3.5 w-3.5" /> Retry

  1087→  // Guard: empty state (no periods configured yet)

  1089→    return (
  1090→      <div className="glass rounded-[20px] p-10 flex flex-col items-center justify-center text-center min-h-[400px]">
  1091→        <Database className="h-10 w-10 text-sky-300 mb-3" />

  1098→          <Plus className="h-3.5 w-3.5" /> Configure Period
  1099→        </button>

  1101→    )
  1102→  }
  1103→
  1104→  const isEmpty = !overview || trendsArr.length === 0
  1105→

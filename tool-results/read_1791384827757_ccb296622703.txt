    59→function statusPillClass(status: string): string {
    60→  const s = (status || '').toUpperCase()
    61→  if (['APPROVED', 'LOCKED', 'SUBSIDIARY_APPROVED', 'BU_APPROVED', 'HQ_REVIEW'].includes(s)) return 'status-approved'
    62→  if (['DRAFT'].includes(s)) return 'status-draft'
    63→  if (['SUBMITTED', 'RESUBMITTED', 'UNDER_REVIEW'].includes(s)) return 'status-submitted'
    64→  if (['REJECTED', 'CORRECTION_REQUESTED', 'MISSING'].includes(s)) return 'status-missing'
    65→  if (['VERIFIED'].includes(s)) return 'status-verified'
    66→  return 'status-locked'
    67→}
    68→
    69→function flattenProjects(tree: OrgTree | null) {
    70→  const out: Array<{
    71→    id: string; projectCode: string; projectName: string; location: string | null
    72→    status: string; groupCode: string; subsidiaryCode: string; buCode: string; buName: string
    73→  }> = []
    74→  if (!tree) return out
    75→  for (const g of tree.groups) {
    76→    for (const sub of g.subsidiaries) {
    77→      for (const bu of sub.businessUnits) {
    78→        for (const p of bu.projects) {
    79→          out.push({
    80→            id: p.id, projectCode: p.projectCode, projectName: p.projectName,
    81→            location: p.location, status: p.status,
    82→            groupCode: g.code, subsidiaryCode: sub.code, buCode: bu.code, buName: bu.name,
    83→          })
    84→        }
    85→      }
    86→    }
    87→  }
    88→  return out
    89→}

   405→/* ---------- Module helpers ---------- */
   406→function modulePct(subs: SubmissionItem[], moduleKey: string): number {
   407→  const sub = subs.find(s => s.module === moduleKey)
   408→  if (!sub) return 0
   409→  return sub.completionPct || 0
   410→}

   412→/* ---------- Sub-components ---------- */
   413→function KpiCard({ icon: Icon, tileClass, label, value, unit, sub, delay }: {

   515→function ActivityTab({ activity, projectId }: { activity: ActivityItem[]; projectId: string | null }) {
   516→  const [loading, setLoading] = useState(false)
   517→  const [items, setItems] = useState<ActivityItem[]>(activity)
   518→  useEffect(() => { setItems(activity) }, [activity, projectId])
   519→  const reload = async () => {
   520→    if (!projectId) return
   521→    setLoading(true)
   522→    try {
   523→      const r = await fetch(`/api/activity?projectId=${encodeURIComponent(projectId)}&take=20`)
   524→      const d = await r.json()
   525→      setItems(d.items || [])
   526→    } finally { setLoading(false) }
   527→  }
   528→  return (
   529→    <div>
   530→      <div className="mb-2 flex items-center justify-between">
   531→        <span className="text-[11px] text-slate-500">Latest events on this project</span>
   532→        <button onClick={reload} disabled={loading}
   533→          className="glass-subtle rounded-full px-2.5 py-1 text-[10px] font-medium text-slate-600 transition hover:bg-white/80">
   534→          {loading ? 'Refreshing…' : 'Refresh'}
   535→        </button>
   536→      </div>
   537→      <div className="max-h-72 space-y-1 overflow-y-auto scroll-elegant pr-1">
   538→        {items.length === 0 && (
   539→          <div className="px-3 py-8 text-center text-[11px] text-slate-400">No recent activity for this project.</div>

   727→function timeAgo(iso: string) {
   728→  const diff = Date.now() - new Date(iso).getTime()
   729→  const mins = Math.floor(diff / 60000)
   730→  if (mins < 1) return 'just now'
   731→  if (mins < 60) return `${mins}m ago`
   732→  const hrs = Math.floor(mins / 60)
   733→  if (hrs < 24) return `${hrs}h ago`
   734→  return `${Math.floor(hrs / 24)}d ago`
   735→}
